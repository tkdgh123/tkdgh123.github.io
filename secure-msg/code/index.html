/* ================= ① 통신: 길이 붙인 패킷 =================
   TCP는 "어디까지가 한 메시지"인지 모른다. 그래서 보낼 때 맨 앞에 길이 4바이트를 붙이고,
   받을 때는 그 길이만큼 다 모일 때까지 기다렸다가 한 통으로 꺼낸다 (conn_take).

   패킷 모양은 암호화 버전(브랜치 with-crypto)과 똑같이 맞춰 두었다. 그래서 지금은 안 쓰는
   칸이 있다: seq·nonce 는 0, HELLO 본문 128바이트는 0으로 채워 보낸다.
   나중에 암호화를 다시 얹을 때 서버 쪽 코드를 거의 안 바꿔도 되게 하려는 것이다. */

#ifndef MSG_NOSIGNAL        /* 맥에는 없다. 대신 main에서 SIGPIPE를 무시한다 */
#define MSG_NOSIGNAL 0
#endif

#define NAME_LEN   16       /* 이름 최대 15바이트 (한글 5글자) + 끝 표시 */
#define NONCE_LEN  24
#define HDR_LEN    (1 + NAME_LEN + NAME_LEN + 8 + NONCE_LEN)   /* 65 */
#define MAX_BODY   65536
#define HELLO_LEN  (32 + 32 + 64)   /* 암호화 버전: 신원 공개키 + 세션 공개키 + 서명. 지금은 0 */
#define PKT_MAX    (4 + HDR_LEN + MAX_BODY)

/* 패킷 종류. 지금 쓰는 건 HELLO(접속하며 이름 밝히기) · MSG(대화) · SYS(서버 알림).
   VAULT~PROOF 는 암호화 버전의 금고용 — 서버는 내용을 안 보고 to 에게 그대로 전한다 */
enum { T_HELLO = 1, T_MSG, T_VAULT, T_OPEN_REQ, T_SHARE, T_PROOF, T_SYS };

struct packet {
    uint8_t  type;
    char     from[NAME_LEN], to[NAME_LEN];   /* 보낸 사람, 받을 사람 */
    uint64_t seq;                            /* 암호화 버전: 순서 번호. 지금은 0 */
    uint8_t  nonce[NONCE_LEN];               /* 암호화 버전: 한 번만 쓰는 값. 지금은 0 */
    uint32_t body_len;
    uint8_t  body[MAX_BODY];
};

struct conn {                 /* 받는 중인 바이트를 모아 두는 곳 */
    int     fd;
    uint8_t buf[PKT_MAX];
    size_t  len;
};

/* port번에서 접속을 기다리는 소켓을 만든다. 실패하면 -1 */
int net_listen(int port)
{
    int fd = socket(AF_INET, SOCK_STREAM, 0);
    if (fd < 0) {
        perror("socket");
        return -1;
    }

    /* 껐다 바로 켜도 Address already in use가 안 나게 */
    int on = 1;
    setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &on, sizeof(on));

    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(port);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);   /* 0.0.0.0: 다른 노트북에서도 접속 가능 */

    if (bind(fd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
        perror("bind");
        close(fd);
        return -1;
    }
    if (listen(fd, 16) < 0) {
        perror("listen");
        close(fd);
        return -1;
    }
    return fd;
}

/* host:port에 접속한다. host는 IP나 이름. 실패하면 -1 */
int net_connect(const char *host, int port)
{
    struct addrinfo hints = {0}, *res;
    char port_str[8];
    hints.ai_family = AF_INET;
    hints.ai_socktype = SOCK_STREAM;
    snprintf(port_str, sizeof(port_str), "%d", port);
    if (getaddrinfo(host, port_str, &hints, &res) != 0) {
        fprintf(stderr, "%s: 주소를 찾을 수 없음\n", host);
        return -1;
    }

    int fd = socket(AF_INET, SOCK_STREAM, 0);
    if (fd < 0) {
        perror("socket");
        freeaddrinfo(res);
        return -1;
    }

    /* 5초 안에 안 붙으면 포기한다. 중간에서 막힌 네트워크면 그냥 두면 2분 넘게 아무 말 없이 멈춘다.
       그래서 잠깐 기다리지 않는 모드로 접속을 걸어 두고, poll로 5초만 기다린다 */
    int flags = fcntl(fd, F_GETFL, 0);
    fcntl(fd, F_SETFL, flags | O_NONBLOCK);
    int r = connect(fd, res->ai_addr, res->ai_addrlen);
    if (r < 0 && errno == EINPROGRESS) {
        struct pollfd p = { .fd = fd, .events = POLLOUT };
        int err = ETIMEDOUT;
        socklen_t len = sizeof(err);
        if (poll(&p, 1, 5000) == 1)
            getsockopt(fd, SOL_SOCKET, SO_ERROR, &err, &len);
        errno = err;
        r = err ? -1 : 0;
    }
    fcntl(fd, F_SETFL, flags);
    freeaddrinfo(res);
    if (r < 0) {
        int e = errno;   /* 왜 실패했는지는 부른 쪽에서 errno로 본다 */
        perror("connect");
        close(fd);
        errno = e;
        return -1;
    }
    return fd;
}

/* 패킷을 길이 4바이트 + 헤더 + body 바이트로 늘어놓고, 보낼 바이트 수를 돌려준다.
   구조체는 패딩이 끼므로 그대로 보내지 않고 한 칸씩 복사한다 */
size_t pack_packet(const struct packet *p, uint8_t *out)
{
    uint32_t len = HDR_LEN + p->body_len;
    uint32_t nlen = htonl(len);
    uint8_t *q = out;

    memcpy(q, &nlen, 4);             q += 4;
    *q++ = p->type;
    memcpy(q, p->from, NAME_LEN);    q += NAME_LEN;
    memcpy(q, p->to, NAME_LEN);      q += NAME_LEN;
    for (int i = 7; i >= 0; i--)      /* seq: 큰 자리부터 한 바이트씩 */
        *q++ = (uint8_t)(p->seq >> (i * 8));
    memcpy(q, p->nonce, NONCE_LEN);  q += NONCE_LEN;
    memcpy(q, p->body, p->body_len);
    return 4 + len;
}

int send_packet(int fd, const struct packet *p)
{
    static uint8_t out[PKT_MAX];
    size_t total = pack_packet(p, out), sent = 0;

    while (sent < total) {           /* send는 일부만 보낼 수 있다 */
        ssize_t n = send(fd, out + sent, total - sent, MSG_NOSIGNAL);
        if (n <= 0)
            return -1;
        sent += (size_t)n;
    }
    return 0;
}

/* 소켓에서 읽은 만큼 버퍼 뒤에 붙인다. >0 받음, 0 끊김, -1 오류 */
int conn_read(struct conn *c)
{
    ssize_t n = recv(c->fd, c->buf + c->len, sizeof(c->buf) - c->len, 0);
    if (n < 0)
        return -1;
    c->len += n;
    return (int)n;
}

/* 버퍼에 패킷이 통째로 모였으면 하나 꺼낸다. 1 꺼냄, 0 덜 옴, -1 잘못된 길이 */
int conn_take(struct conn *c, struct packet *p)
{
    uint32_t len;
    if (c->len < 4)
        return 0;
    memcpy(&len, c->buf, 4);
    len = ntohl(len);
    if (len < HDR_LEN || len > HDR_LEN + MAX_BODY)
        return -1;
    if (c->len < 4 + len)
        return 0;                    /* 아직 덜 왔음: 다음 recv를 기다림 */

    const uint8_t *q = c->buf + 4;
    p->type = *q++;
    memcpy(p->from, q, NAME_LEN);    q += NAME_LEN;
    memcpy(p->to, q, NAME_LEN);      q += NAME_LEN;
    p->from[NAME_LEN - 1] = p->to[NAME_LEN - 1] = '\0';
    p->seq = 0;
    for (int i = 0; i < 8; i++)
        p->seq = (p->seq << 8) | *q++;
    memcpy(p->nonce, q, NONCE_LEN);  q += NONCE_LEN;
    p->body_len = len - HDR_LEN;
    memcpy(p->body, q, p->body_len);

    c->len -= 4 + len;               /* 꺼낸 만큼 앞으로 당긴다 */
    memmove(c->buf, c->buf + 4 + len, c->len);
    return 1;
}

/* 터미널 폭 cols에서 s를 찍으면 줄이 몇 번 넘어가나 (한글은 두 칸) */
static int rows_after(const char *s, int cols)
{
    mbstate_t st;
    wchar_t wc;
    size_t len = strlen(s), n;
    int x = 0, rows = 0;

    memset(&st, 0, sizeof(st));
    while (len > 0) {
        n = mbrtowc(&wc, s, len, &st);
        if (n == (size_t)-2)
            break;
        int cw = 1;
        if (n == (size_t)-1) {
            memset(&st, 0, sizeof(st));
            n = 1;
        } else if ((cw = wcwidth(wc)) < 0) {
            cw = 0;
        }
        if (x + cw > cols) {
            rows++;
            x = 0;
        }
        x += cw;
        s += n;
        len -= n;
    }
    return rows;
}

#define C_ACCENT (tty ? "\033[36m" : "")
#define C_RESET  (tty ? "\033[0m"  : "")

static void clear_input(void)
{
    if (!tty)
        return;
    if (drawn_rows > 0)
        printf("\033[%dA", drawn_rows);
    printf("\r\033[J");
}

static void draw_input(void)
{
    if (!tty)
        return;
    char line[64 + INPUT_MAX];
    struct winsize ws;
    int cols = 80;

    prompt(line, sizeof(line));
    strcat(line, input);
    printf("%s", line);
    if (ioctl(1, TIOCGWINSZ, &ws) == 0 && ws.ws_col > 0)
        cols = ws.ws_col;
    drawn_rows = rows_after(line, cols);
    fflush(stdout);
}

/* 치던 입력 줄을 잠깐 지우고 한 줄 찍은 뒤 다시 그린다. 맨 앞 [태그]는 청록 */
static void put_line(const char *text)
{
    clear_input();
    if (tty && text[0] == '[') {
        const char *close = strchr(text, ']');
        if (close)
            printf("\033[36m%.*s\033[0m%s\n", (int)(close - text + 1), text, close + 1);
        else
            printf("%s\n", text);
    } else
        printf("%s\n", text);
    draw_input();
    fflush(stdout);
}

static void show(const char *fmt, ...)
{
    char text[2100];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(text, sizeof(text), fmt, ap);
    va_end(ap);

}

static void clean(char *s)   /* 받은 글자 중 제어 문자는 ? 로 */
{
    for (; *s; s++)
        if ((unsigned char)*s < 0x20 || *s == 0x7f)
            *s = '?';
}

static int valid_name(const char *s)
{
    size_t n = strlen(s);
    if (n == 0 || n >= NAME_LEN)
        return 0;
    for (; *s; s++)
        if (*s == ' ' || *s == '/' || (unsigned char)*s < 0x20)
            return 0;
    return 1;
}

/* 대화 화면에서 아는 사람들 (방장·참가자 공통). /who, /dm, 전체 보내기에 쓴다 */
struct peer { char name[NAME_LEN]; };
static struct peer peers[MAX_CLIENTS];
static int npeers;

static int find_peer(const char *name)
{
    for (int i = 0; i < npeers; i++)
        if (strncmp(peers[i].name, name, NAME_LEN) == 0)
            return i;
    return -1;
}

static void add_peer(const char *name)
{
    if (name[0] == '\0' || strncmp(name, my_name, NAME_LEN) == 0
        || find_peer(name) >= 0 || npeers >= MAX_CLIENTS)
        return;
    strncpy(peers[npeers++].name, name, NAME_LEN - 1);
}

static void remove_peer(const char *name)
{
    int i = find_peer(name);
    if (i < 0)
        return;
    if (strncmp(dm_to, name, NAME_LEN) == 0)
        dm_to[0] = '\0';
    peers[i] = peers[--npeers];
}

/* ================= ③ 중계: 서버와 방장이 같이 쓴다 ================= */

struct client {
    struct conn c;
    char    name[NAME_LEN];
    uint8_t hello[HELLO_LEN];
    int     registered;
};

/* 한 칸 더: 17번째의 HELLO까지 읽은 뒤에 "꽉 참"을 보내야 알림이 사라지지 않는다
   (안 읽은 데이터가 남은 채 close하면 RST가 나가서 받는 쪽이 알림을 못 읽는다) */
static struct client clients[MAX_CLIENTS + 1];
static int nclients;

static const char *type_name(int type)
{
    static const char *names[] = { "?", "HELLO", "MSG", "VAULT", "OPEN_REQ", "SHARE", "PROOF", "SYS" };
    return (type >= T_HELLO && type <= T_SYS) ? names[type] : names[0];
}

static int find_client(const char *name)
{
    for (int i = 0; i < nclients; i++)
        if (clients[i].registered && strncmp(clients[i].name, name, NAME_LEN) == 0)
            return i;
    return -1;
}

static void pkt_init(struct packet *p, uint8_t type, const char *from, const char *to)
{
    memset(p, 0, sizeof(*p));
    p->type = type;
    strncpy(p->from, from, NAME_LEN - 1);
    if (to)
        strncpy(p->to, to, NAME_LEN - 1);
}

/* i번에게 알림 한 줄 */
static void send_sys(int i, const char *fmt, ...)
{
    va_list ap;
    pkt_init(&out, T_SYS, "", clients[i].name);
    va_start(ap, fmt);
    int n = vsnprintf((char *)out.body, MAX_BODY, fmt, ap);
    va_end(ap);
    out.body_len = n < MAX_BODY ? n : MAX_BODY - 1;
    send_packet(clients[i].c.fd, &out);
}

/* name의 HELLO를 i번에게 */
static void send_hello(int i, const char *name, const uint8_t *hello)
{
    pkt_init(&out, T_HELLO, name, clients[i].name);
    out.body_len = HELLO_LEN;
    memcpy(out.body, hello, HELLO_LEN);
    send_packet(clients[i].c.fd, &out);
}

static void add_client(int fd)
{
    if (fd < 0)
        return;
    if (nclients == MAX_CLIENTS + 1) {   /* 남는 칸까지 찼으면 말없이 닫는다 */
        close(fd);
        return;
    }
    memset(&clients[nclients], 0, sizeof(clients[0]));
    clients[nclients].c.fd = fd;
    nclients++;
}

/* i번을 지우고 마지막 칸을 i번 자리로 옮긴다 */
static void drop_client(int i)
{
    char name[NAME_LEN];
    int was_registered = clients[i].registered;
    memcpy(name, clients[i].name, NAME_LEN);

    close(clients[i].c.fd);
    clients[i] = clients[--nclients];

    if (was_registered) {
        show("[알림] %s 퇴장", name);
        remove_peer(name);
        for (int k = 0; k < nclients; k++)
            if (clients[k].registered)
                send_sys(k, "%s 퇴장", name);
    }
}

static int handle_hello(int i, const struct packet *p)
{
    if (clients[i].registered) {
        send_sys(i, "이미 HELLO를 보냈습니다");
        return 0;
    }
    if (p->body_len != HELLO_LEN) {
        send_sys(i, "HELLO 길이가 잘못됨");
        drop_client(i);
        return -1;
    }
    if (!valid_name(p->from) || find_client(p->from) >= 0
        || (mode == M_HOST && strncmp(p->from, my_name, NAME_LEN) == 0)) {
        send_sys(i, "이름 사용 중이거나 쓸 수 없음: %s", p->from);
        drop_client(i);
        return -1;
    }
    int registered = 0;
    for (int k = 0; k < nclients; k++)
        registered += clients[k].registered;
    if (registered == MAX_CLIENTS) {
        send_sys(i, "방이 꽉 찼습니다 (%d명)", MAX_CLIENTS);
        drop_client(i);
        return -1;
    }

    struct client *me = &clients[i];
    memcpy(me->name, p->from, NAME_LEN);
    memcpy(me->hello, p->body, HELLO_LEN);
    me->registered = 1;
    show("[알림] %s 입장", me->name);

    for (int k = 0; k < nclients; k++) {
        if (k == i || !clients[k].registered)
            continue;
        send_hello(i, clients[k].name, clients[k].hello);   /* 새 사람에게 기존 전원의 HELLO */
        send_hello(k, me->name, me->hello);                 /* 기존 전원에게 새 HELLO */
    }
    if (mode == M_HOST) {   /* 방장도 방에 있는 한 사람이다 */
        static const uint8_t no_key[HELLO_LEN];
        send_hello(i, my_name, no_key);
        add_peer(me->name);
    }
    for (int k = 0; k < nclients; k++)
        if (clients[k].registered)
            send_sys(k, "%s 입장", me->name);
    return 0;
}

/* 서버의 눈: 중계할 때마다 한 줄. 평문 단계라 내용이 그대로 보인다 (방장은 대화 화면이라 안 찍음) */
static void show_eye(const struct packet *p)
{
    if (mode != M_SERVER)
        return;
    char line[200];
    int off = snprintf(line, sizeof(line), "[relay] %s → %s  %-8s seq=%-4llu %4uB ", p->from, p->to,
                       type_name(p->type), (unsigned long long)p->seq, p->body_len);
    for (uint32_t k = 0; k < p->body_len && k < 8; k++)
        off += snprintf(line + off, sizeof(line) - off, " %02x", p->body[k]);
    show("%s%s", line, p->body_len > 8 ? " …" : "");
}

/* to에게 중계 */
static void relay(int to, struct packet *p)
{
    show_eye(p);
    send_packet(clients[to].c.fd, p);   /* 실패해도 다음 poll에서 끊김으로 잡힌다 */
}

static void display_msg(const struct packet *p);   /* ④에 있음 */
static void show_guide(void);                      /* ⑤ 바로 위에 있음 */

/* 패킷 하나 처리. 이 사람을 끊었으면 -1 */
static int handle_packet(int i, struct packet *p)
{
    if (!clients[i].registered && p->type != T_HELLO) {
        send_sys(i, "먼저 HELLO를 보내야 합니다");
        drop_client(i);
        return -1;
    }
    if (p->type == T_HELLO)
        return handle_hello(i, p);
    if (p->type < T_MSG || p->type > T_PROOF) {
        send_sys(i, "모르는 type: %d", p->type);
        return 0;
    }
    if (strncmp(p->from, clients[i].name, NAME_LEN) != 0) {
        send_sys(i, "보낸 사람 이름이 다릅니다: %s", p->from);   /* 남의 이름으로 보내기 차단 */
        return 0;
    }
    if (mode == M_HOST && strncmp(p->to, my_name, NAME_LEN) == 0) {
        if (p->type == T_MSG)
            display_msg(p);   /* 방장한테 온 것 */
        return 0;
    }

    int to = find_client(p->to);
    if (to < 0) {
        send_sys(i, "%s는 접속해 있지 않습니다", p->to);
        return 0;
    }
    relay(to, p);
    return 0;
}

/* i번 소켓에 온 것을 읽어서 패킷마다 처리 */
static void serve(int i)
{
    if (conn_read(&clients[i].c) <= 0) {
        drop_client(i);
        return;
    }
    int r;
    while ((r = conn_take(&clients[i].c, &pkt)) == 1)
        if (handle_packet(i, &pkt) < 0)
            return;   /* 이미 drop_client(i) 됨 */
    if (r < 0)
        drop_client(i);   /* 길이가 잘못됨: 버퍼를 늘리지 않고 끊는다 */
}

/* ================= ④ 대화: 방장·참가자 ================= */

static struct conn srv;   /* 참가자: 서버(또는 방장)와의 연결 */

static void display_msg(const struct packet *p)
{
    char from[NAME_LEN + 1];
    static char text[MAX_BODY];
    if (p->body_len < 1)
        return;
    memcpy(from, p->from, NAME_LEN);
    from[NAME_LEN] = '\0';
    clean(from);
    size_t n = p->body_len - 1;
    if (n > MAX_BODY - 1)
        n = MAX_BODY - 1;
    memcpy(text, p->body + 1, n);
    text[n] = '\0';
    clean(text);
    if (p->body[0] == TO_ONE)
        show("[%s → 나] %s", from, text);
    else
        show("[%s] %s", from, text);
}

/* SYS "이름 입장/퇴장" 이면 명단을 고친다 (참가자 쪽) */
static void sys_update(const char *text)
{
    size_t L = strlen(text);
    const char *in = " 입장", *ou = " 퇴장";
    size_t li = strlen(in), lo = strlen(ou);
    char nm[NAME_LEN];
    if (L > li && strcmp(text + L - li, in) == 0 && L - li < NAME_LEN) {
        memcpy(nm, text, L - li);
        nm[L - li] = '\0';
        add_peer(nm);
    } else if (L > lo && strcmp(text + L - lo, ou) == 0 && L - lo < NAME_LEN) {
        memcpy(nm, text, L - lo);
        nm[L - lo] = '\0';
        remove_peer(nm);
    }
}

/* 참가자: 서버(방장)에게서 받기 */
static void join_read(void)
{
    if (conn_read(&srv) <= 0) {
        show("[알림] 연결이 끊어졌습니다");
        exit(1);
    }
    int r;
    while ((r = conn_take(&srv, &pkt)) == 1) {
        if (pkt.type == T_HELLO) {
            clean(pkt.from);
            add_peer(pkt.from);
        } else if (pkt.type == T_MSG) {
            display_msg(&pkt);
        } else if (pkt.type == T_SYS) {
            static char text[MAX_BODY];
            size_t n = pkt.body_len < MAX_BODY ? pkt.body_len : MAX_BODY - 1;
            memcpy(text, pkt.body, n);
            text[n] = '\0';
            clean(text);
            show("[알림] %s", text);
            sys_update(text);
        }
        /* 평문 버전에는 금고·증명 같은 다른 type이 오지 않는다 */
    }
    if (r < 0) {
        show("[알림] 이상한 패킷이 와서 끊습니다");
        exit(1);
    }
}

/* 한 사람에게 보낸다. 방장은 직접 중계하고, 참가자는 서버(방장)에게 맡긴다 */
static void send_one(const char *to, uint8_t flag, const char *text)
{
    size_t len = strlen(text);
    if (len > INPUT_MAX)
        len = INPUT_MAX;
    pkt_init(&out, T_MSG, my_name, to);
    out.body[0] = flag;
    memcpy(out.body + 1, text, len);
    out.body_len = (uint32_t)(len + 1);

    if (mode == M_HOST) {
        int k = find_client(to);
        if (k >= 0)
            relay(k, &out);
    } else {
        send_packet(srv.fd, &out);
    }
}

/* 전체 대화는 사람마다 한 통씩 */
static void send_text(const char *text)
{
    if (dm_to[0])
        send_one(dm_to, TO_ONE, text);
    else if (npeers == 0)
        show("[알림] 아직 아무도 없습니다");
    else
        for (int i = 0; i < npeers; i++)
            send_one(peers[i].name, TO_ALL, text);
}

static void cmd_who(void)
{
    char line[64 + MAX_CLIENTS * (NAME_LEN + 2)];
    int off = snprintf(line, sizeof(line), "[접속 중] %s(나%s)", my_name, mode == M_HOST ? "·방장" : "");
    for (int i = 0; i < npeers; i++)
        off += snprintf(line + off, sizeof(line) - off, ", %s", peers[i].name);
    show("%s", line);
}

static void echo_line(const char *line)   /* 친 줄을 화면에 남긴다 */
{
    if (!tty)
        return;
    char p[64];
    prompt(p, sizeof(p));
    clear_input();
    printf("%s%s\n", p, line);
    draw_input();
    fflush(stdout);
}

static void on_line(const char *line)
{
    if (strcmp(line, "/quit") == 0) {
        clear_input();
        exit(0);
    } else if (strcmp(line, "/all") == 0) {
        dm_to[0] = '\0';
        show("[전체 대화] 접속한 사람 모두에게 갑니다");
    } else if (strncmp(line, "/dm ", 4) == 0) {
        const char *name = line + 4;
        while (*name == ' ')   /* "/dm  친구" 처럼 띄어쓰기를 두 번 해도 */
            name++;
        if (strcmp(name, my_name) == 0 || !name[0])
            show("[1:1 대화] 쓰는 법: /dm 이름");
        else if (find_peer(name) < 0)
            show("[1:1 대화] %s는 접속해 있지 않습니다 (/who 로 확인)", name);
        else {
            strncpy(dm_to, name, NAME_LEN - 1);
            show("[1:1 대화] %s에게만 갑니다 · /all 로 돌아오기", name);
        }
    } else if (strcmp(line, "/who") == 0) {
        cmd_who();
    } else if (strcmp(line, "/help") == 0) {
        show_guide();
    } else if (line[0] == '/') {
        show("[알림] 모르는 명령입니다 · /help");
    } else {
        send_text(line);
    }
}

/* 앞뒤 공백을 지운다. 복사해 붙이면 끝에 공백이나 NBSP(눈에 안 보이는 공백, C2 A0)가 딸려 오기 쉽다 */
static char *trim(char *s)
{
    for (;;) {
        if (*s == ' ' || *s == '\t')
            s++;
        else if ((unsigned char)s[0] == 0xC2 && (unsigned char)s[1] == 0xA0)
            s += 2;
        else
            break;
    }
    size_t n = strlen(s);
    for (;;) {
        if (n > 0 && (s[n - 1] == ' ' || s[n - 1] == '\t'))
            n--;
        else if (n > 1 && (unsigned char)s[n - 2] == 0xC2 && (unsigned char)s[n - 1] == 0xA0)
            n -= 2;
        else
            break;
    }
    s[n] = '\0';
    return s;
}

/* 키 하나. 엔터면 한 줄 처리, 지우기·Ctrl+U·Ctrl+C·Ctrl+D, 화살표 같은 ESC 줄은 무시 */
static void on_key(unsigned char ch)
{
    static int esc;
    if (esc == 1) {
        esc = (ch == '[') ? 2 : 0;
        return;
    }
    if (esc == 2) {
        if (ch >= 0x40 && ch <= 0x7e)
            esc = 0;
        return;
    }
    if (ch == 0x1b) {
        esc = 1;
    } else if (ch == '\n' || ch == '\r') {
        char line[INPUT_MAX + 1];
        if (input_len == 0)
            return;
        strcpy(line, input);
        input_len = 0;
        input[0] = '\0';
        char *text = trim(line);   /* "/who " 처럼 끝에 공백이 붙어도 명령으로 알아듣게 */
        if (text[0] == '\0')
            return;
        echo_line(text);
        on_line(text);
    } else if (ch == 0x7f || ch == 0x08) {
        while (input_len > 0 && (input[input_len - 1] & 0xC0) == 0x80)
            input_len--;   /* 한글은 여러 바이트: 글자 하나를 통째로 지운다 */
        if (input_len > 0)
            input_len--;
        input[input_len] = '\0';
    } else if (ch == 0x03 || (ch == 0x04 && input_len == 0)) {
        clear_input();
        exit(0);
    } else if (ch == 0x15) {
        input_len = 0;
        input[0] = '\0';
    } else if (ch >= 0x20) {
        int lead = (ch & 0xC0) != 0x80;
        if (input_len >= INPUT_MAX || (lead && input_len >= INPUT_MAX - 4))
            return;
        input[input_len++] = (char)ch;
        input[input_len] = '\0';
    }
}

static void read_keys(void)
{
    unsigned char buf[256];
    ssize_t n = read(0, buf, sizeof(buf));
    if (n <= 0) {
        clear_input();
        exit(0);
    }
    for (ssize_t i = 0; i < n; i++)
        on_key(buf[i]);
    clear_input();
    draw_input();
}

static void restore_tty(void)
{
    if (tty)
        tcsetattr(0, TCSANOW, &saved_tio);
}

/* 키를 누르는 즉시 받도록 (엔터 기다리지 않음, 화면에 저절로 찍지 않음) */
static void raw_tty(void)
{
    tty = isatty(0) && isatty(1);
    if (!tty)
        return;
    tcgetattr(0, &saved_tio);
    struct termios t = saved_tio;
    t.c_lflag &= ~(ICANON | ECHO | ISIG);
    t.c_cc[VMIN] = 1;
    t.c_cc[VTIME] = 0;
    tcsetattr(0, TCSANOW, &t);
    atexit(restore_tty);
}

/* 방을 연 쪽(서버·방장): 다른 컴퓨터가 들어올 때 칠 내 주소를 보여 준다.
   루프백(127.0.0.1 같은 자기 자신용)과 169.254.x(주소를 못 받았을 때 임시)는 뺀다 */
static void show_my_address(void)
{
    struct ifaddrs *list;
    char ip[INET_ADDRSTRLEN], others[160] = "";
    int n = 0;

    if (getifaddrs(&list) == 0) {
        for (struct ifaddrs *a = list; a; a = a->ifa_next) {
            if (!a->ifa_addr || a->ifa_addr->sa_family != AF_INET
                || (a->ifa_flags & IFF_LOOPBACK) || !(a->ifa_flags & IFF_UP))
                continue;
            inet_ntop(AF_INET, &((struct sockaddr_in *)a->ifa_addr)->sin_addr, ip, sizeof(ip));
            if (strncmp(ip, "169.254.", 8) == 0)
                continue;
            if (n++ == 0)
                show("[주소] %s — 다른 컴퓨터는  ./deolmi 이름 %s  으로 들어와요", ip, ip);
            else
                snprintf(others + strlen(others), sizeof(others) - strlen(others), " %s", ip);
            int b = atoi(ip + 4);   /* 172.16~31 은 WSL 안쪽 주소일 수 있다 */
            if (strncmp(ip, "172.", 4) == 0 && b >= 16 && b <= 31)
                show("[주소] 172.로 시작하면 WSL 안쪽 주소일 수 있어요 — 미러링을 켜야 밖에서 들어와요 (시작하기 05)");
        }
        freeifaddrs(list);
    }
    if (others[0])
        show("[주소] 안 되면 이 주소로:%s", others);
    if (n == 0)
        show("[주소] 네트워크 주소를 못 찾았어요 — 같은 컴퓨터에서는  ./deolmi 이름  으로 들어와요");
}

/* 시작할 때와 /help 때 보여 주는 설명. 처음 쓰는 사람도 이것만 보고 쓸 수 있게 */
static void show_guide(void)
{
    const char *rule = "────────────────────────────────────────────";
    show("%s", rule);
    if (mode == M_SERVER) {

        show_my_address();

        if (mode == M_HOST) {

        show("사용법");
        show("  그냥 치고 엔터   방에 있는 모두에게 보내기");
        show("  /dm 이름         그 사람하고만 1:1  (/all 로 다시 모두에게)");

        show("받은 글은 [이름] 내용, 나에게만 온 1:1은 [이름 → 나] 내용 으로 떠요.");
    }
    show("지금은 암호화 전 단계라 메시지가 평문(그대로)으로 오가요.");
    show("만든 사람: 김창식(기획·결정·실행 확인) · Claude(AI — 코드와 사이트 작성)");

/* ================= ⑤ 실행 ================= */

int main(int argc, char **argv)
{
    setlocale(LC_ALL, "");
    signal(SIGPIPE, SIG_IGN);
    setvbuf(stdout, NULL, _IOLBF, 0);   /* 파일로 돌려도 줄마다 바로 나오게 */

    int lfd = -1;
    if (argc == 2 && strcmp(argv[1], "--server") == 0) {
        mode = M_SERVER;
        lfd = net_listen(PORT);
        if (lfd < 0)
            return 1;
        show("[서버] 기다리는 중");
        show_guide();
    } else if (argc >= 2 && argv[1][0] != '-' && valid_name(argv[1])) {
        strncpy(my_name, argv[1], NAME_LEN - 1);
        mode = (argc >= 3 && strcmp(argv[2], "--host") == 0) ? M_HOST : M_JOIN;
        if (mode == M_HOST) {
            lfd = net_listen(PORT);
            if (lfd < 0)
                return 1;
            raw_tty();
            show("%s덜미%s · %s · 방장", C_ACCENT, C_RESET, my_name);
        } else {
            const char *host = argc >= 3 ? argv[2] : "127.0.0.1";
            printf("%s 에 접속하는 중… (5초)\n", host);
            srv.fd = net_connect(host, PORT);
            if (srv.fd < 0) {
                if (errno == ECONNREFUSED)
                    fprintf(stderr, "→ 그 주소에 열린 방이 없어요. 방장이 먼저 --host 로 방을 열었는지, 주소가 맞는지 보세요\n");
                else if (errno == ETIMEDOUT || errno == EHOSTUNREACH || errno == ENETUNREACH)
                    fprintf(stderr, "→ 연결이 중간에서 막혔어요. 방장 노트북 방화벽, 같은 네트워크인지 보세요.\n"
                                    "  학교 와이파이는 기기끼리 연결을 막는 경우가 많아요 — 둘 다 휴대폰 핫스팟에 연결해 보세요\n");
                return 1;
            }
            pkt_init(&out, T_HELLO, my_name, NULL);
            out.body_len = HELLO_LEN;
            send_packet(srv.fd, &out);
            raw_tty();
            show("%s덜미%s · %s · %s 에 접속", C_ACCENT, C_RESET, my_name, host);
        }
        show_guide();
    } else {
        fprintf(stderr, "덜미 — 터미널 채팅 (프밍기2 팀 프로젝트)\n\n");
        fprintf(stderr, "  %s 이름 --host       방을 연다 (방장: 서버 없이 내가 중계하면서 대화)\n", argv[0]);
        fprintf(stderr, "  %s 이름 [주소]       방에 들어간다 (주소는 방장 화면의 [주소] 줄, 같은 컴퓨터면 생략)\n", argv[0]);
        fprintf(stderr, "  %s --server          중계만 하는 서버를 켠다 (대화는 안 함)\n\n", argv[0]);
        fprintf(stderr, "이름은 한글 5글자(15바이트)까지, 띄어쓰기와 / 없이.\n");
        fprintf(stderr, "만든 사람: 김창식(기획·결정·실행 확인) · Claude(AI — 코드와 사이트 작성)\n");
        fprintf(stderr, "자세한 안내: https://tkdgh123.github.io/secure-msg/run/\n");
        return 1;
    }

    int key_fd = mode == M_SERVER ? -1 : 0;   /* 키보드: 서버는 안 본다 (-1이면 poll이 건너뜀) */
    struct pollfd fds[2 + MAX_CLIENTS + 1];
    for (;;) {

/* s를 cols칸 터미널에 찍으면 몇 줄 아래로 넘어가나.
   한글은 2칸이고, 줄 끝에 1칸만 남으면 터미널이 그 글자를 다음 줄로 넘긴다 */
static int rows_after(const char *s, int cols)
{
    mbstate_t st;
    wchar_t wc;
    size_t len = strlen(s), n;
    int x = 0, rows = 0;

    memset(&st, 0, sizeof(st));
    while (len > 0) {
        n = mbrtowc(&wc, s, len, &st);
        if (n == (size_t)-2)
            break;                      /* 아직 덜 들어온 글자는 화면에 안 나온다 */
        int cw = 1;
        if (n == (size_t)-1) {          /* 깨진 바이트는 1칸 */
            memset(&st, 0, sizeof(st));
            n = 1;
        } else if ((cw = wcwidth(wc)) < 0) {
            cw = 0;
        }
        if (x + cw > cols) {
            rows++;
            x = 0;
        }
        x += cw;
        s += n;
        len -= n;
    }
    return rows;
}

/* 화면 색 (터미널일 때만). 입력 프롬프트에는 쓰지 않는다 — rows_after가
   ESC 바이트를 칸으로 세어 줄 계산이 틀어진다. 메시지 줄에만 쓴다 */
#define C_ACCENT (tty ? "\033[36m" : "")   /* 이름·단어 */
#define C_RESET  (tty ? "\033[0m"  : "")

static void clear_input(void)
{
    if (!tty)
        return;
    if (drawn_rows > 0)
        printf("\033[%dA", drawn_rows);   /* 입력 줄 첫 줄로 올라가서 */
    printf("\r\033[J");                     /* 그 아래를 다 지운다 */
}

static void draw_input(void)
{
    if (!tty)
        return;
    char line[64 + INPUT_MAX];
    struct winsize ws;
    int cols = 80;

    prompt(line, sizeof(line));
    strcat(line, input);
    printf("%s", line);
    if (ioctl(1, TIOCGWINSZ, &ws) == 0 && ws.ws_col > 0)
        cols = ws.ws_col;
    drawn_rows = rows_after(line, cols);
    fflush(stdout);
}

/* 입력 줄을 지우고 한 줄 찍은 다음 입력 줄을 다시 그린다. 경고는 빨갛게 */
static void put_line(int warning, const char *text)
{
    clear_input();
    if (warning && tty)
        printf("\033[31m%s\033[0m\n", text);
    else if (tty && text[0] == '[') {   /* 줄 맨 앞 [태그]를 강조 */
        const char *close = strchr(text, ']');
        if (close)
            printf("\033[36m%.*s\033[0m%s\n", (int)(close - text + 1), text, close + 1);
        else
            printf("%s\n", text);
    } else
        printf("%s\n", text);
    draw_input();
    fflush(stdout);
}

static void log_line(const char *conv, const char *text)
{
    char line[2100];
    time_t t = time(NULL);
    struct tm *tm = localtime(&t);
    if (!history_on())
        return;
    snprintf(line, sizeof(line), "[%02d:%02d] %s", tm->tm_hour, tm->tm_min, text);
    history_add(conv, line);
}

static void show(const char *fmt, ...)
{
    char text[2100];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(text, sizeof(text), fmt, ap);
    va_end(ap);
    put_line(0, text);
}

/* 경고는 대화 기록에도 남긴다 */
static void warn(const char *fmt, ...)
{
    char text[2100];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(text, sizeof(text), fmt, ap);
    va_end(ap);
    put_line(1, text);
    log_line("all", text);
}

/* 남이 보낸 글에 섞인 제어 문자를 ?로. 터미널 조작 문자(ESC 등)를 막는다 */
static void clean(char *s)
{
    for (; *s; s++)
        if ((unsigned char)*s < 0x20 || *s == 0x7f)
            *s = '?';
}

static void restore_tty(void)
{
    if (tty)
        tcsetattr(0, TCSANOW, &saved_tio);
}

static void raw_tty(void)
{
    tty = isatty(0) && isatty(1);
    if (!tty)
        return;
    tcgetattr(0, &saved_tio);
    struct termios t = saved_tio;
    t.c_lflag &= ~(ICANON | ECHO | ISIG);   /* 한 글자씩 받고, 화면은 직접 그리고, Ctrl+C도 직접 */
    t.c_cc[VMIN] = 1;
    t.c_cc[VTIME] = 0;
    tcsetattr(0, TCSANOW, &t);
    atexit(restore_tty);
}

static void to_hex(const uint8_t *b, size_t n, char *out)
{
    for (size_t i = 0; i < n; i++)
        sprintf(out + 2 * i, "%02x", b[i]);
}

static int from_hex(const char *s, uint8_t *b, size_t n)
{
    if (strlen(s) != 2 * n)
        return -1;
    for (size_t i = 0; i < n; i++)
        if (sscanf(s + 2 * i, "%2hhx", &b[i]) != 1)
            return -1;
    return 0;
}

/* known 파일 한 줄 = 이름 신원공개키(hex) 확인여부 */
static void known_load(void)
{
    char name[64], hex[2 * KEY_LEN + 2];
    int v;
    FILE *f = fopen(known_path, "r");
    if (!f)
        return;
    while (nknown < MAX_KNOWN && fscanf(f, "%63s %65s %d", name, hex, &v) == 3) {
        if (strlen(name) >= NAME_LEN || from_hex(hex, known[nknown].pk, KEY_LEN) != 0)
            continue;
        strcpy(known[nknown].name, name);
        known[nknown].verified = v;
        nknown++;
    }
    fclose(f);
}

static void known_save(void)
{
    char hex[2 * KEY_LEN + 1];
    FILE *f = fopen(known_path, "w");
    if (!f)
        return;
    for (int i = 0; i < nknown; i++) {
        to_hex(known[i].pk, KEY_LEN, hex);
        fprintf(f, "%s %s %d\n", known[i].name, hex, known[i].verified);
    }
    fclose(f);
}

static int find_known(const char *name)
{
    for (int i = 0; i < nknown; i++)
        if (strcmp(known[i].name, name) == 0)
            return i;
    return -1;
}

static void known_set(const char *name, const uint8_t pk[KEY_LEN], int verified)
{
    int k = find_known(name);
    if (k < 0) {
        if (nknown == MAX_KNOWN)
            return;
        k = nknown++;
        strcpy(known[k].name, name);
    }
    memcpy(known[k].pk, pk, KEY_LEN);
    known[k].verified = verified;
    known_save();
}

/* 이미 승인한 금고 번호 (같은 요청을 다시 받으면 거절) */
static int approved_has(const uint8_t id[KEY_LEN])
{
    char line[128], hex[2 * KEY_LEN + 1];
    int found = 0;
    FILE *f = fopen(approved_path, "r");
    to_hex(id, KEY_LEN, hex);
    while (f && !found && fgets(line, sizeof(line), f))
        found = strncmp(line, hex, 2 * KEY_LEN) == 0;
    if (f)
        fclose(f);
    return found;
}

static void approved_add(const uint8_t id[KEY_LEN])
{
    char hex[2 * KEY_LEN + 1];
    FILE *f = fopen(approved_path, "a");
    if (!f)
        return;
    to_hex(id, KEY_LEN, hex);
    fprintf(f, "%s\n", hex);
    fclose(f);
}

/* 이름에 쓸 수 없는 것: 빈 이름, 15바이트 넘음, 제어 문자, 띄어쓰기, /, 맨 앞 . (파일 이름에 쓰므로) */
static int valid_name(const char *s)
{
    if (s[0] == '\0' || s[0] == '.' || strlen(s) >= NAME_LEN)
        return 0;
    for (; *s; s++)
        if ((unsigned char)*s < 0x20 || *s == 0x7f || *s == ' ' || *s == '/')
            return 0;
    return 1;
}

static void on_hello(const struct packet *p)
{
    char name[NAME_LEN], old_words[WORDS_LEN];
    uint8_t old_pk[KEY_LEN];
    int had_old = 0;

    memcpy(name, p->from, NAME_LEN);
    if (p->body_len != HELLO_LEN || !valid_name(name) || strcmp(name, my_name) == 0)
        return;
    if (hello_check(name, p->body) != 0) {   /* 신원 키가 kx 키에 찍은 도장이 맞나 */
        warn("[!] %s의 HELLO 서명이 맞지 않아 무시합니다", name);
        return;
    }

    int i = find_peer(name), k = find_known(name);
    if (i >= 0) {                  /* 같은 세션에서 또 HELLO: 다시 접속했거나 서버의 바꿔치기 */
        memcpy(old_pk, peers[i].id_pk, KEY_LEN);
        had_old = 1;
    } else if (k >= 0) {           /* 전에 만난 적 있음 */
        memcpy(old_pk, known[k].pk, KEY_LEN);
        had_old = 1;
    }
    if (i < 0) {
        if (npeers == MAX_PEERS)
            return;
        i = npeers++;
    }

    /* 새 HELLO = 새 접속: 열쇠도 번호도 처음부터 */
    struct peer *pe = &peers[i];
    memset(pe, 0, sizeof(*pe));
    memcpy(pe->name, name, NAME_LEN);
    memcpy(pe->id_pk, p->body, KEY_LEN);
    fingerprint(pe->id_pk, pe->words);
    pe->has_keys = kx_chains(my_kx_pk, my_kx_sk, p->body + KX_OFFSET, &pe->tx, &pe->rx) == 0;
    if (!pe->has_keys)
        warn("[!] %s의 열쇠가 잘못돼서 이 사람과는 대화할 수 없습니다", name);

    if (had_old && memcmp(old_pk, pe->id_pk, KEY_LEN) != 0) {
        fingerprint(old_pk, old_words);
        warn("[!] %s의 열쇠가 바뀜: %s → %s", name, old_words, pe->words);
        warn("    %s에게 단어를 소리 내어 다시 읽어 달라고 하세요 (/verify %s)", name, name);
        known_set(name, pe->id_pk, 0);
    } else if (k < 0) {
        show("[열쇠] %s 처음 만남 · 단어 %s · 만나서 맞춰 보고 /verify %s ok", name, pe->words, name);
        known_set(name, pe->id_pk, 0);
    } else if (known[k].verified) {
        show("[열쇠] %s ✔ 전에 단어를 확인한 열쇠", name);
    } else {
        show("[열쇠] %s · 단어 %s (아직 확인 안 함 · /verify %s)", name, pe->words, name);
    }
    k = find_known(name);
    pe->verified = k >= 0 && known[k].verified;
}

/* AD = 헤더의 type · from · to · seq. 서버가 헤더를 바꾸면 복호화가 실패한다 */
static void make_ad(uint8_t ad[AD_LEN], const struct packet *p)
{
    ad[0] = p->type;
    memcpy(ad + 1, p->from, NAME_LEN);
    memcpy(ad + 1 + NAME_LEN, p->to, NAME_LEN);
    for (int i = 0; i < 8; i++)
        ad[1 + 2 * NAME_LEN + i] = (uint8_t)(p->seq >> (56 - 8 * i));
}

static void send_simple(int type, const char *to, const uint8_t *body, size_t len, const uint8_t *nonce)
{
    out.type = (uint8_t)type;
    strncpy(out.from, my_name, NAME_LEN);
    strncpy(out.to, to, NAME_LEN);
    out.seq = 0;
    if (nonce)
        memcpy(out.nonce, nonce, NONCE_LEN);
    else
        memset(out.nonce, 0, NONCE_LEN);
    memcpy(out.body, body, len);
    out.body_len = (uint32_t)len;
    send_packet(srv.fd, &out);
}

static int send_sealed(int i, int type, const uint8_t *pt, size_t len)
{
    struct peer *pe = &peers[i];
    uint8_t ad[AD_LEN];

    if (!pe->has_keys) {
        warn("[!] %s와는 열쇠가 없어서 보내지 않았습니다", pe->name);
        return -1;
    }
    out.type = (uint8_t)type;
    strncpy(out.from, my_name, NAME_LEN);
    strncpy(out.to, pe->name, NAME_LEN);
    out.seq = ++pe->send_seq;
    make_ad(ad, &out);
    out.body_len = (uint32_t)chain_seal(&pe->tx, ad, pt, len, out.nonce, out.body);
    return send_packet(srv.fd, &out);
}

static void send_msg(int i, int kind, const char *text)
{
    static uint8_t pt[INPUT_MAX + 2];
    size_t len = strlen(text);
    pt[0] = (uint8_t)kind;
    memcpy(pt + 1, text, len);
    send_sealed(i, T_MSG, pt, 1 + len);
}

static void show_proof(const struct proof *pr)
{
    uint8_t packed[PROOF_LEN];
    char words[WORDS_LEN], title[TITLE_LEN];
    time_t t = (time_t)pr->time;
    struct tm *tm = localtime(&t);

    proof_pack(pr, packed);
    words_of(packed, PROOF_LEN, 3, words, sizeof(words));   /* 선택 3: 셋이 카톡에 붙여 넣어 대조 */
    memcpy(title, pr->title, TITLE_LEN);
    clean(title);
    snprintf(last_proof, sizeof(last_proof),
             "[증명] %s가 %s에게 '%s' 열쇠를 넘김 · %s의 금고 · %02d:%02d · %s✔ %s✔ · 증명 단어 %s%s",
             pr->appr, pr->req, title, pr->owner, tm->tm_hour, tm->tm_min, pr->req, pr->appr, words,
             strcmp(pr->owner, my_name) == 0 ? " · 내가 맡긴 금고" : "");
    show("%s", last_proof);
    log_line("all", last_proof);
}

static void on_proof(const struct peer *pe, const uint8_t *pt, int n)
{
    struct proof pr;
    if (proof_unpack(&pr, pt, n) != 0 || !valid_name(pr.req) || !valid_name(pr.appr) || !valid_name(pr.owner)) {
        warn("[!] %s가 보낸 증명의 모양이 이상함 · 무시", pe->name);
        return;
    }
    if (strcmp(pr.req, pe->name) != 0) {
        warn("[!] %s가 남(%s)의 증명을 보냄 · 무시", pe->name, pr.req);
        return;
    }
    int j = find_peer(pr.appr);
    const uint8_t *appr_pk = strcmp(pr.appr, my_name) == 0 ? my_id_pk : j >= 0 ? peers[j].id_pk : NULL;
    if (!appr_pk) {
        warn("[!] 승인한 %s가 접속해 있지 않아 증명을 확인할 수 없습니다", pr.appr);
        return;
    }
    if (proof_check(&pr, pe->id_pk, appr_pk) != 0) {
        warn("[!] 증명 서명 불일치: %s가 보낸 증명 · 무시", pe->name);
        return;
    }
    show_proof(&pr);
}

/* 받는 쪽이 순서번호를 검사하고, 풀리면 래칫을 한 칸 전진 */
static void on_sealed(const struct packet *p)
{
    static uint8_t pt[MAX_BODY];
    static char text[MAX_BODY];
    char from[NAME_LEN], conv[NAME_LEN + 4];
    static char line[MAX_BODY + 64];
    uint8_t ad[AD_LEN];
    unsigned long long seq = p->seq;

    make_ad(ad, p);
    memcpy(from, p->from, NAME_LEN);
    clean(from);

    int i = find_peer(p->from);
    if (i < 0 || !peers[i].has_keys) {
        warn("[!] 열쇠를 모르는 %s가 보낸 메시지 · 폐기", from);
        return;
    }
    struct peer *pe = &peers[i];
    if (seq <= pe->recv_seq) {
        warn("[!] 순서번호 불일치: %s의 #%llu 메시지는 이미 받은 번호 (재전송 의심) · 폐기", from, seq);
        return;
    }
    if (pe->broken) {
        warn("[!] %s와의 대화가 끊겨서 #%llu 메시지를 열 수 없습니다. 다시 접속하세요", from, seq);
        return;
    }
    if (seq > pe->recv_seq + 1) {   /* 건너뛸 수 없다: 빠진 문장을 몰라 다음 열쇠를 못 만든다 */
        pe->broken = 1;
        warn("[!] %s→나 #%llu 메시지를 못 받아서 그 뒤로는 열 수 없습니다: 대화가 끊겼습니다. 다시 접속하세요",
             from, (unsigned long long)pe->recv_seq + 1);
        return;
    }

    int n = chain_open(&pe->rx, ad, p->nonce, p->body, p->body_len, pt);
    if (n < 0) {
        warn("[!] 변조 감지: %s의 #%llu 메시지가 잠근 그대로가 아님 (중간에 바뀜) · 폐기", from, seq);
        return;
    }
    pe->recv_seq = seq;
    if (p->type == T_PROOF) {
        on_proof(pe, pt, n);
        return;
    }
    if (n < 1 || n - 1 > INPUT_MAX)
        return;

    memcpy(text, pt + 1, n - 1);
    text[n - 1] = '\0';
    clean(text);
    if (pt[0] == TO_ONE) {
        snprintf(conv, sizeof(conv), "dm-%s", from);
        snprintf(line, sizeof(line), "[%s → 나] %s", from, text);
    } else {
        strcpy(conv, "all");
        snprintf(line, sizeof(line), "[%s] %s", from, text);
    }
    show("%s", line);
    log_line(conv, line);

    if (pe->has_stolen) {   /* 선택 2: 훔친 체인 키는 앞으로 오는 메시지를 연다 */
        static uint8_t pt2[MAX_BODY];
        int m = chain_open(&pe->stolen, ad, p->nonce, p->body, p->body_len, pt2);
        if (m > 1) {
            pt2[m] = '\0';
            clean((char *)pt2 + 1);
            warn("[탈취 실험] 훔친 열쇠로 #%llu 메시지도 열림: %s", seq, (char *)pt2 + 1);
            warn("    그 뒤 메시지도 같은 식으로 열립니다. 새로 접속해서 열쇠를 바꿔야 막힙니다");
        }
        wipe(&pe->stolen, sizeof(pe->stolen));
        pe->has_stolen = 0;
    }
    if (p->body_len <= sizeof(pe->past[0].ct)) {   /* 지난 암호문을 몇 개 남겨 둔다 */
        struct past *ps = &pe->past[pe->npast++ % 4];
        ps->seq = seq;
        memcpy(ps->ad, ad, AD_LEN);
        memcpy(ps->nonce, p->nonce, NONCE_LEN);
        memcpy(ps->ct, p->body, p->body_len);
        ps->len = p->body_len;
    }
}

    return k;
}

/* 가진 금고를 하나씩 vbuf에 불러 본다. want가 있으면 그 제목에서 멈추고 1 */
static int scan_vaults(const char *want, char *path_out, size_t size)
{
    char path[1024], title[TITLE_LEN], shown[TITLE_LEN];
    uint8_t piece[PIECE_LEN];
    int count = 0;
    DIR *d = opendir(vault_dir);
    struct dirent *e;

    while (d && (e = readdir(d)) != NULL) {
        size_t n = strlen(e->d_name);
        if (n < 7 || strcmp(e->d_name + n - 6, ".vault") != 0)
            continue;
        snprintf(path, sizeof(path), "%s/%s", vault_dir, e->d_name);
        if (vault_load(path, &vbuf) != 0) {
            warn("[!] 금고 파일이 바뀌어 서명이 안 맞음: %s", e->d_name);
            continue;
        }
        int k = my_piece(&vbuf, piece, title);
        wipe(piece, sizeof(piece));
        if (k < 0)
            continue;
        if (want) {
            if (strcmp(title, want) == 0) {
                snprintf(path_out, size, "%s", path);
                closedir(d);
                return 1;
            }
            continue;
        }
        memcpy(shown, title, TITLE_LEN);
        clean(shown);
        show("[금고] '%s' · %s가 맡김 · 함께 가진 사람 %s", shown, vbuf.owner, vault_holder(&vbuf, 1 - k));
        count++;
    }
    if (d)
        closedir(d);
    if (!want && count == 0)
        show("[금고] 가지고 있는 금고가 없습니다");
    return 0;
}

/* /lock B C "제목" 내용 */
static void cmd_lock(const char *args)
{
    char h[2][64], title[TITLE_LEN + 1];
    int off = 0, idx[2];

    if (sscanf(args, "%63s %63s %n", h[0], h[1], &off) < 2 || args[off] != '"') {
        show("[금고] 쓰는 법: /lock B C \"제목\" 내용");
        return;
    }
    const char *t = args + off + 1, *q = strchr(t, '"');
    if (!q || q == t || q - t >= TITLE_LEN) {
        show("[금고] 제목은 따옴표 안에 1~31바이트 (한글 10글자)");
        return;
    }
    memcpy(title, t, q - t);
    title[q - t] = '\0';
    const char *body = q + 1;
    while (*body == ' ')
        body++;
    if (*body == '\0' || strlen(body) > MAX_SECRET) {
        show("[금고] 내용을 1~%d바이트로 적어 주세요", MAX_SECRET);
        return;
    }
    if (strcmp(h[0], h[1]) == 0 || strcmp(h[0], my_name) == 0 || strcmp(h[1], my_name) == 0) {
        show("[금고] 나 말고 서로 다른 두 사람에게 맡깁니다");
        return;
    }
    for (int k = 0; k < 2; k++) {
        idx[k] = find_peer(h[k]);
        if (idx[k] < 0 || !peers[idx[k]].has_keys) {
            show("[금고] %s가 접속해 있어야 맡길 수 있습니다", h[k]);
            return;
        }
    }

    if (vault_make(&vbuf, my_name, my_id_pk, my_id_sk, h[0], peers[idx[0]].id_pk,
                   h[1], peers[idx[1]].id_pk, title, body) != 0) {
        show("[금고] 금고를 만들지 못했습니다");
        return;
    }
    for (int k = 0; k < 2; k++)
        send_simple(T_VAULT, h[k], vbuf.body, vbuf.len, vbuf.nonce);
    show("[금고] '%s' 금고를 %s, %s에게 맡겼습니다. 둘이 함께여야 열리고, 이 노트북에는 남기지 않습니다",
         title, h[0], h[1]);
    for (int k = 0; k < 2; k++)
        if (!peers[idx[k]].verified)
            show("[*] %s의 단어를 아직 확인하지 않았습니다. 서버가 열쇠를 바꿔치기했다면 그 조각은 서버가 엽니다", h[k]);
    wipe(&vbuf, sizeof(vbuf));
}

static void on_vault(const struct packet *p)
{
    char path[600], title[TITLE_LEN];
    uint8_t piece[PIECE_LEN];
    int i = find_peer(p->from);

    if (i < 0) {
        warn("[!] 모르는 사람이 보낸 금고 · 버림");
        return;
    }
    memset(&vbuf, 0, sizeof(vbuf));
    memcpy(vbuf.owner, p->from, NAME_LEN);
    memcpy(vbuf.owner_pk, peers[i].id_pk, KEY_LEN);   /* HELLO에서 받은 그 사람의 신원 키로 확인 */
    memcpy(vbuf.nonce, p->nonce, NONCE_LEN);
    vbuf.len = p->body_len;
    memcpy(vbuf.body, p->body, p->body_len);

    if (vault_verify(&vbuf) != 0) {
        warn("[!] 서명 불일치: %s가 만든 금고가 아님 (위조) · 버림", p->from);
        return;
    }
    int k = my_piece(&vbuf, piece, title);
    wipe(piece, sizeof(piece));
    if (k < 0) {
        warn("[!] %s가 보낸 금고에 내 조각이 없거나 열리지 않음 · 버림", p->from);
        return;
    }
    vault_path(vault_dir, vbuf.id, path, sizeof(path));
    if (access(path, F_OK) == 0 || approved_has(vbuf.id)) {
        warn("[!] 이미 받은 금고가 또 옴 (재전송 의심) · 무시");
        return;
    }
    if (vault_save(vault_dir, &vbuf) != 0) {
        warn("[!] 금고를 저장하지 못했습니다");
        return;
    }
    clean(title);
    show("[금고] %s가 '%s' 금고를 맡겼습니다 · 함께 가진 사람 %s · 열려면 /open %s",
         p->from, title, vault_holder(&vbuf, 1 - k), title);
    if (!peers[i].verified)
        show("[*] %s의 단어를 아직 확인하지 않았습니다 (/verify %s)", p->from, p->from);
}

/* /open 제목: 함께 가진 사람에게 서명한 열람 요청 */
static void cmd_open(const char *title)
{
    char path[600];
    uint8_t body[OPEN_REQ_LEN];

    if (!scan_vaults(title, path, sizeof(path))) {
        show("[금고] '%s' 금고가 없습니다 (/vault 로 목록)", title);
        return;
    }
    int k = vault_holder_index(&vbuf, my_name);
    const char *other = vault_holder(&vbuf, 1 - k);
    int j = find_peer(other);
    if (j < 0 || !peers[j].has_keys) {
        show("[금고] %s가 접속해 있어야 열 수 있습니다", other);
        return;
    }
    open_req_make(body, vbuf.id, my_name, other, my_id_sk);
    send_simple(T_OPEN_REQ, other, body, sizeof(body), NULL);
    pending_open.active = 1;
    memcpy(pending_open.id, vbuf.id, KEY_LEN);
    snprintf(pending_open.who, NAME_LEN, "%s", other);
    show("[*] %s의 승인을 기다리는 중", other);
}

static void on_open_req(const struct packet *p)
{
    char path[600], title[TITLE_LEN];
    uint8_t piece[PIECE_LEN];
    int i = find_peer(p->from);

    if (i < 0 || p->body_len != OPEN_REQ_LEN) {
        warn("[!] 알 수 없는 열람 요청 · 거절");
        return;
    }
    if (approved_has(p->body)) {
        warn("[!] 이미 승인한 금고를 %s가 다시 요청함 (재전송 의심) · 거절", p->from);
        return;
    }
    vault_path(vault_dir, p->body, path, sizeof(path));
    if (vault_load(path, &vbuf) != 0) {
        warn("[!] 내게 없는 금고를 %s가 열려고 함 · 거절", p->from);
        return;
    }
    int k = my_piece(&vbuf, piece, title);
    wipe(piece, sizeof(piece));
    if (k < 0 || vault_holder_index(&vbuf, p->from) != 1 - k) {
        warn("[!] %s는 이 금고를 함께 가진 사람이 아님 · 거절", p->from);
        return;
    }
    if (open_req_check(p->body, p->body_len, p->from, my_name, peers[i].id_pk) != 0) {
        warn("[!] 열람 요청 서명 불일치 (%s가 보낸 게 아님) · 거절", p->from);
        return;
    }
    pending_req.active = 1;
    memcpy(pending_req.id, vbuf.id, KEY_LEN);
    snprintf(pending_req.who, NAME_LEN, "%s", p->from);
    clean(title);
    show("[요청] %s가 %s의 '%s' 금고를 열려고 합니다 · 승인하려면 /approve", p->from, vbuf.owner, title);
    if (!peers[i].verified)
        show("[*] %s의 단어를 아직 확인하지 않았습니다. 승인하면 조각이 지금 %s라고 하는 열쇠로 봉인됩니다",
             p->from, p->from);
}

/* /approve: 내 조각을 요청한 사람에게만 다시 봉인 + 증명에 서명 → 내 사본은 지움 */
static void cmd_approve(void)
{
    char path[600], title[TITLE_LEN];
    uint8_t piece[PIECE_LEN], body[SHARE_PKT_LEN];
    struct proof pr;

    if (!pending_req.active) {
        show("[금고] 승인할 요청이 없습니다");
        return;
    }
    int j = find_peer(pending_req.who);
    vault_path(vault_dir, pending_req.id, path, sizeof(path));
    if (j < 0 || !peers[j].has_keys) {
        show("[금고] %s가 접속해 있지 않습니다", pending_req.who);
        return;
    }
    if (vault_load(path, &vbuf) != 0 || my_piece(&vbuf, piece, title) < 0) {
        show("[금고] 금고를 열지 못했습니다");
        return;
    }
    proof_fill(&pr, &vbuf, title, (uint64_t)time(NULL), pending_req.who, my_name);
    proof_sign(&pr, my_id_sk, pr.appr_sig);
    int ok = share_make(body, &pr, piece, peers[j].id_pk) == 0;
    wipe(piece, sizeof(piece));
    if (!ok) {
        show("[금고] 조각을 봉인하지 못했습니다");
        return;
    }
    send_simple(T_SHARE, pending_req.who, body, sizeof(body), NULL);
    unlink(path);
    approved_add(pending_req.id);
    pending_req.active = 0;
    clean(title);
    show("[금고] %s에게 '%s' 열쇠 조각을 넘겼습니다. 내 사본은 지웠습니다", pending_req.who, title);
}

static void show_secret(const char *title)
{

    show("    Enter를 누르거나 %d초가 지나면 화면에서 지웁니다", SHOW_SECS);

    secret_until = time(NULL) + SHOW_SECS;
}

static void on_share(const struct packet *p)
{
    char path[600], title[TITLE_LEN];
    uint8_t mine[PIECE_LEN], theirs[PIECE_LEN], packed[PROOF_LEN];
    struct proof pr;
    int i = find_peer(p->from);

    if (i < 0 || !pending_open.active || strcmp(p->from, pending_open.who) != 0
        || p->body_len < KEY_LEN || memcmp(p->body, pending_open.id, KEY_LEN) != 0) {
        warn("[!] 요청하지 않은 승인 조각이 %s에게서 옴 · 무시", p->from);
        return;
    }
    vault_path(vault_dir, pending_open.id, path, sizeof(path));
    if (vault_load(path, &vbuf) != 0 || my_piece(&vbuf, mine, title) < 0) {
        warn("[!] 내 금고를 불러오지 못했습니다");
        return;
    }
    int r = share_open(p->body, p->body_len, &vbuf, my_name, my_id_pk, my_id_sk, p->from,
                       peers[i].id_pk, theirs, &pr);
    if (r == 0 && memcmp(theirs + KEY_LEN, title, TITLE_LEN) != 0)
        r = -3;
    if (r == 0 && vault_reveal(&vbuf, mine, theirs, secret) < 0)
        r = -4;
    wipe(mine, sizeof(mine));
    wipe(theirs, sizeof(theirs));
    if (r == -1)
        warn("[!] %s가 보낸 승인 조각이 열리지 않음 · 버림", p->from);
    else if (r == -2)
        warn("[!] 승인 서명 불일치: %s가 승인한 게 아님 · 버림", p->from);
    else if (r < 0)
        warn("[!] 두 조각으로도 금고가 열리지 않음 (조각이 바뀜) · 버림");
    if (r < 0)
        return;

    unlink(path);
    pending_open.active = 0;
    clean(title);
    show_secret(title);
    proof_sign(&pr, my_id_sk, pr.req_sig);
    show_proof(&pr);
    proof_pack(&pr, packed);
    for (int k = 0; k < npeers; k++)   /* 증명을 접속한 사람 모두에게 (제목이 있어서 래칫으로 잠근다) */
        if (peers[k].has_keys)
            send_sealed(k, T_PROOF, packed, PROOF_LEN);
}

static void handle_packet(struct packet *p)
{
    static char text[MAX_BODY + 1];

    switch (p->type) {
    case T_HELLO:
        on_hello(p);
        break;
    case T_MSG:
    case T_PROOF:
        on_sealed(p);
        break;
    case T_VAULT:
        on_vault(p);
        break;
    case T_OPEN_REQ:
        on_open_req(p);
        break;
    case T_SHARE:
        on_share(p);
        break;
    case T_SYS:
        memcpy(text, p->body, p->body_len);
        text[p->body_len] = '\0';
        clean(text);
        show("[알림] %s", text);
        for (int i = 0; i < npeers; i++) {   /* "B 퇴장"이면 목록에서 뺀다 */
            char left[NAME_LEN + 16];
            snprintf(left, sizeof(left), "%s 퇴장", peers[i].name);
            if (strcmp(text, left) == 0) {
                remove_peer(i);
                break;
            }
        }
        break;
    }
}

static void cmd_verify(const char *args)
{
    char name[64] = "", ok[8] = "";

        return;
    }
    struct peer *pe = &peers[i];

        pe->verified = 1;

        show("[열쇠] %s ✔ 확인한 열쇠로 기억합니다", pe->name);
        return;
    }
    show("[%s의 단어] %s%s", pe->name, pe->words, pe->verified ? " ✔ 확인함" : "");
    show("    %s 화면의 '내 단어'와 같으면 /verify %s ok", pe->name, pe->name);
}

/* 선택 2: 지금 받는 체인 키를 훔쳤다고 치고, 지난 메시지와 다음 메시지를 열어 본다 */
static void cmd_steal(const char *name)
{
    static uint8_t pt[MAX_BODY];
    char hex[17];
    int i = find_peer(name);
    if (i < 0 || !peers[i].has_keys) {
        show("[알림] %s는 접속해 있지 않습니다", name);
        return;
    }
    struct peer *pe = &peers[i];
    pe->stolen = pe->rx;
    pe->has_stolen = 1;
    to_hex(pe->stolen.ck, 8, hex);
    warn("[탈취 실험] %s에게서 받는 체인 키를 훔쳤다고 칩니다: %s…", name, hex);
    int tried = 0;
    for (int k = 0; k < 4 && k < pe->npast; k++) {
        struct chain c = pe->stolen;
        struct past *ps = &pe->past[k];
        int n = chain_open(&c, ps->ad, ps->nonce, ps->ct, ps->len, pt);
        show("    지난 #%llu 메시지: %s", ps->seq, n < 0 ? "못 엶 ✔ (지난 메시지는 안전)" : "열림 (!)");
        wipe(&c, sizeof(c));
        tried++;
    }
    if (tried == 0)
        show("    아직 %s에게서 받은 메시지가 없어 지난 메시지로 시험할 게 없습니다", name);
    show("    대신 %s가 다음에 보내는 메시지는 이 열쇠로 열립니다", name);
}

static void on_line(const char *line)
{
    char conv[NAME_LEN + 4], logged[INPUT_MAX + 64], p[64];

    if (strcmp(line, "/quit") == 0) {
        exit(0);
    } else if (strcmp(line, "/all") == 0) {
        dm_to[0] = '\0';
        show("[전체 대화] 접속한 사람 모두에게 갑니다");
    } else if (strncmp(line, "/dm ", 4) == 0) {
        const char *name = line + 4;
        if (strcmp(name, my_name) == 0)
            show("[알림] 나에게는 보낼 수 없습니다");
        else if (find_peer(name) < 0)
            show("[알림] %s는 접속해 있지 않습니다", name);
        else {
            snprintf(dm_to, sizeof(dm_to), "%s", name);
            show("[1:1 대화] %s에게만 갑니다 · /all 로 돌아오기", dm_to);
        }
    } else if (strcmp(line, "/who") == 0) {
        char list[MAX_PEERS * (NAME_LEN + 8)] = "";
        for (int i = 0; i < npeers; i++) {
            strcat(list, ", ");
            strcat(list, peers[i].name);
            if (peers[i].verified)
                strcat(list, " ✔");
        }
        show("[접속 중] %s(나)%s", my_name, list);
    } else if (strcmp(line, "/verify") == 0 || strncmp(line, "/verify ", 8) == 0) {
        cmd_verify(line + 7);
    } else if (strncmp(line, "/lock ", 6) == 0) {
        cmd_lock(line + 6);
    } else if (strncmp(line, "/open ", 6) == 0) {
        cmd_open(line + 6);
    } else if (strcmp(line, "/approve") == 0) {
        cmd_approve();
    } else if (strcmp(line, "/vault") == 0) {
        scan_vaults(NULL, NULL, 0);
    } else if (strcmp(line, "/history") == 0 || strncmp(line, "/history ", 9) == 0) {
        int count = atoi(line + 8);
        if (dm_to[0])
            snprintf(conv, sizeof(conv), "dm-%s", dm_to);
        else
            strcpy(conv, "all");
        history_show(conv, count > 0 ? count : 20, history_out);
    } else if (strcmp(line, "/forget") == 0) {
        history_forget();
        show("[기록] 대화 기록과 salt를 지웠습니다. 기록이 꺼졌습니다");
    } else if (strncmp(line, "/steal ", 7) == 0) {
        cmd_steal(line + 7);
    } else if (line[0] == '/') {
        show("[알림] 대화: /dm 이름 · /all · /who · 열쇠: /verify [이름 [ok]]");
        show("       금고: /lock B C \"제목\" 내용 · /open 제목 · /approve · /vault");
        show("       기록: /history [줄 수] · /forget · 실험: /steal 이름 · /quit");
    } else {
        prompt(p, sizeof(p));
        snprintf(logged, sizeof(logged), "%s%s", p, line);
        if (dm_to[0]) {
            send_msg(find_peer(dm_to), TO_ONE, line);
            snprintf(conv, sizeof(conv), "dm-%s", dm_to);
        } else if (npeers == 0) {
            show("[알림] 아직 아무도 없습니다");
            return;
        } else {
            for (int i = 0; i < npeers; i++)   /* 전체 대화: 사람마다 따로 보낸다 */
                send_msg(i, TO_ALL, line);
            strcpy(conv, "all");
        }
        log_line(conv, logged);
    }
}

/* 친 줄을 화면에 남긴다. /lock 의 내용은 가린다 */
static void echo_line(const char *line)
{

    addr.sin_port = htons(port);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);   /* 0.0.0.0: 다른 노트북에서도 접속 가능 */

static uint8_t buf[16 + NAME_LEN + NONCE_LEN + MAX_BODY + SIG_LEN];   /* 서명할 내용을 이어 붙이는 곳 */

/* 서명하는 내용 = "deolmi-vault" ‖ 맡긴 사람 ‖ nonce ‖ 서명을 뺀 body */
static size_t vault_signed(const struct vault *v)
{
    size_t n = 0;
    memcpy(buf, "deolmi-vault", 12);        n += 12;
    memcpy(buf + n, v->owner, NAME_LEN);    n += NAME_LEN;
    memcpy(buf + n, v->nonce, NONCE_LEN);   n += NONCE_LEN;
    memcpy(buf + n, v->body, v->len - SIG_LEN);
    return n + v->len - SIG_LEN;
}

/* 맡긴 사람의 서명을 확인하고 금고 번호를 계산한다. 서명이 틀리면 -1 */
int vault_verify(struct vault *v)
{
    if (v->len < VAULT_FIXED + TAG_LEN + SIG_LEN || v->len > MAX_BODY)
        return -1;
    size_t n = vault_signed(v);
    const uint8_t *sig = v->body + v->len - SIG_LEN;
    if (id_verify(v->owner_pk, buf, n, sig) != 0)
        return -1;
    memcpy(buf + n, sig, SIG_LEN);
    hash32(buf, n + SIG_LEN, v->id);
    return 0;
}

/* 잠금 키 L로 내용을 잠그고, L = S0 ⊕ S1로 나눠 (조각 + 제목)을 보유자마다 봉인한 뒤 서명.
   L, S0, S1은 만들고 바로 지운다 */
int vault_make(struct vault *v, const char *owner, const uint8_t owner_pk[KEY_LEN],
               const uint8_t owner_sk[ID_SK_LEN], const char *holder0, const uint8_t holder0_pk[KEY_LEN],
               const char *holder1, const uint8_t holder1_pk[KEY_LEN],
               const char *title, const char *secret)
{
    uint8_t lock[KEY_LEN], s[2][KEY_LEN], piece[PIECE_LEN];
    const uint8_t *holder_pk[2] = { holder0_pk, holder1_pk };
    size_t slen = strlen(secret);
    int ok = 0;

    if (strlen(title) >= TITLE_LEN || slen > MAX_SECRET)
        return -1;
    memset(v, 0, sizeof(*v));
    strncpy(v->owner, owner, NAME_LEN - 1);
    memcpy(v->owner_pk, owner_pk, KEY_LEN);

    uint8_t *q = v->body;
    strncpy((char *)q, holder0, NAME_LEN - 1);     q += NAME_LEN;
    strncpy((char *)q, holder1, NAME_LEN - 1);     q += NAME_LEN;

    random_bytes(lock, KEY_LEN);
    random_bytes(s[0], KEY_LEN);
    for (int i = 0; i < KEY_LEN; i++)
        s[1][i] = lock[i] ^ s[0][i];           /* 둘 다 있어야 L이 나온다 */

    memset(piece, 0, PIECE_LEN);
    memcpy(piece + KEY_LEN, title, strlen(title));
    for (int k = 0; k < 2; k++) {
        memcpy(piece, s[k], KEY_LEN);
        if (seal_to(holder_pk[k], piece, PIECE_LEN, q) != 0)
            goto out;
        q += SHARE_LEN;
    }
    q += aead_seal(lock, NULL, 0, (const uint8_t *)secret, slen, v->nonce, q);
    v->len = (uint32_t)(q - v->body) + SIG_LEN;
    id_sign(owner_sk, buf, vault_signed(v), v->body + v->len - SIG_LEN);
    ok = vault_verify(v) == 0;
out:
    wipe(lock, sizeof(lock));
    wipe(s, sizeof(s));
    wipe(piece, sizeof(piece));
    return ok ? 0 : -1;
}

                uint8_t piece[PIECE_LEN])
{

        return -1;

    return 0;
}

/* 두 조각을 XOR해 L을 되살리고 내용을 연다. secret은 MAX_SECRET + 1바이트 */
int vault_reveal(const struct vault *v, const uint8_t key0[KEY_LEN], const uint8_t key1[KEY_LEN],
                 char *secret)
{
    uint8_t lock[KEY_LEN];
    size_t ct_len = v->len - VAULT_FIXED - SIG_LEN;

    if (ct_len < TAG_LEN || ct_len - TAG_LEN > MAX_SECRET)
        return -1;
    for (int i = 0; i < KEY_LEN; i++)
        lock[i] = key0[i] ^ key1[i];
    int n = aead_open(lock, NULL, 0, v->nonce, v->body + VAULT_FIXED, ct_len, (uint8_t *)secret);
    wipe(lock, KEY_LEN);
    if (n < 0)
        return -1;
    secret[n] = '\0';
    return n;
}

int history_start(const char *dir)
{

    uint8_t salt[SALT_LEN], nonce[24], check[CHECK_LEN], plain[CHECK_LEN];

    snprintf(dir_path, sizeof(dir_path), "%s", dir);   /* 꺼져 있어도 /forget은 되게 먼저 */

        if (pw[0] == '\0')
            return 0;

            return 0;
        }
        random_bytes(salt, SALT_LEN);
        if (pw_key(pw, salt, key) != 0)
            return 0;
        aead_seal(key, (const uint8_t *)"check", 5, (const uint8_t *)"deolmi", 6, nonce, check);
        fd = open(path, O_WRONLY | O_CREAT | O_EXCL, 0600);
        if (fd < 0 || write(fd, salt, SALT_LEN) != SALT_LEN || write(fd, nonce, 24) != 24
            || write(fd, check, CHECK_LEN) != CHECK_LEN) {
            printf("[*] 기록 파일을 만들 수 없습니다\n");
            if (fd >= 0)
                close(fd);
            wipe(key, KEY_LEN);
            return 0;
        }
        close(fd);
        on = 1;
        printf("[*] 대화 기록을 켰습니다. 다음에 켤 때 이 비밀번호를 넣으면 /history 로 볼 수 있습니다\n");
    } else {
        int ok = read(fd, salt, SALT_LEN) == SALT_LEN && read(fd, nonce, 24) == 24
              && read(fd, check, CHECK_LEN) == CHECK_LEN;
        close(fd);
        for (int tries = 0; ok && tries < 3 && !on; tries++) {
            read_secret("[*] 대화 기록 비밀번호 (Enter = 이번엔 기록 안 함): ", pw, sizeof(pw));
            if (pw[0] == '\0')
                break;

                && aead_open(key, (const uint8_t *)"check", 5, nonce, check, CHECK_LEN, plain) == 6)
                on = 1;
            else
                printf("[*] 비밀번호가 틀렸습니다\n");
        }
    }
    wipe(pw, sizeof(pw));
    wipe(pw2, sizeof(pw2));
    if (!on)
        wipe(key, KEY_LEN);
    return on;
}

/* AD = 대화 이름 ‖ 줄 번호 8바이트 */
static size_t make_ad(uint8_t *ad, const char *conv, uint64_t line_no)
{
    size_t n = strlen(conv);
    memcpy(ad, conv, n);
    for (int i = 7; i >= 0; i--)
        ad[n++] = (uint8_t)(line_no >> (i * 8));
    return n;
}

/* 줄 하나 읽기. 1 읽음, 0 파일 끝, -1 망가짐 */
static int read_record(int fd, uint8_t nonce[24], uint8_t *ct, uint32_t *ct_len)
{
    uint8_t len4[4];
    ssize_t n = read(fd, len4, 4);
    if (n == 0)
        return 0;
    *ct_len = (uint32_t)len4[0] << 24 | len4[1] << 16 | len4[2] << 8 | len4[3];
    if (n != 4 || *ct_len < TAG_LEN || *ct_len > MAX_LINE + TAG_LEN)
        return -1;
    if (read(fd, nonce, 24) != 24 || read(fd, ct, *ct_len) != (ssize_t)*ct_len)
        return -1;
    return 1;
}

void history_add(const char *conv, const char *line)
{
    static uint8_t rec[4 + 24 + MAX_LINE + TAG_LEN], ct[MAX_LINE + TAG_LEN];
    char path[600], name[64];
    uint8_t ad[64 + 8], nonce[24];
    uint32_t ct_len;
    uint64_t count = 0;
    size_t len = strlen(line);

    if (!on || len > MAX_LINE)
        return;
    snprintf(name, sizeof(name), "%s.log", conv);
    file_path(name, path, sizeof(path));
    int fd = open(path, O_RDWR | O_CREAT | O_APPEND, 0600);
    if (fd < 0)
        return;
    while (read_record(fd, nonce, ct, &ct_len) == 1)   /* 지금까지 몇 줄인지 = 새 줄 번호 */
        count++;

    size_t ad_len = make_ad(ad, conv, count);
    ct_len = (uint32_t)aead_seal(key, ad, ad_len, (const uint8_t *)line, len, rec + 4, rec + 4 + 24);
    rec[0] = (uint8_t)(ct_len >> 24);
    rec[1] = (uint8_t)(ct_len >> 16);
    rec[2] = (uint8_t)(ct_len >> 8);
    rec[3] = (uint8_t)ct_len;
    if (write(fd, rec, 4 + 24 + ct_len) < 0)
        perror("history");
    close(fd);
}

/* 지금 대화의 마지막 count줄을 풀어서 out으로. 안 풀리는 줄은 변조 경고 */
void history_show(const char *conv, int count, void (*out)(int warning, const char *line))
{
    static uint8_t ct[MAX_LINE + TAG_LEN];
    static char plain[MAX_LINE + 1], msg[MAX_LINE + 64];
    char path[600], name[64];
    uint8_t ad[64 + 8], nonce[24];
    uint32_t ct_len;
    uint64_t total = 0, i = 0;

    if (!on) {
        out(0, "[알림] 대화 기록이 꺼져 있습니다 (켤 때 비밀번호를 정하면 켜집니다)");
        return;
    }
    snprintf(name, sizeof(name), "%s.log", conv);
    file_path(name, path, sizeof(path));
    int fd = open(path, O_RDONLY);
    if (fd < 0) {
        out(0, "[기록] 아직 없습니다");
        return;
    }
    int r;
    while ((r = read_record(fd, nonce, ct, &ct_len)) == 1)
        total++;
    if (r < 0) {
        snprintf(msg, sizeof(msg), "[!] 기록 변조 감지: %s #%llu 뒤가 망가짐", name, (unsigned long long)total + 1);
        out(1, msg);
    }
    lseek(fd, 0, SEEK_SET);
    snprintf(msg, sizeof(msg), "[기록] %s · %llu줄 중 마지막 %llu줄", name, (unsigned long long)total,
             (unsigned long long)(total < (uint64_t)count ? total : (uint64_t)count));
    out(0, msg);

    for (i = 0; i < total && read_record(fd, nonce, ct, &ct_len) == 1; i++) {
        if (i + count < total)
            continue;
        int n = aead_open(key, ad, make_ad(ad, conv, i), nonce, ct, ct_len, (uint8_t *)plain);
        if (n < 0) {
            snprintf(msg, sizeof(msg), "[!] 기록 변조 감지: %s #%llu", name, (unsigned long long)i + 1);
            out(1, msg);
            continue;
        }
        plain[n] = '\0';
        for (char *s = plain; *s; s++)   /* 파일에서 온 글도 터미널 조작 문자는 막는다 */
            if ((unsigned char)*s < 0x20 || *s == 0x7f)
                *s = '?';
        out(0, plain);
    }
    close(fd);
}

            unlink(path);
        }
    }
    if (d)

}
