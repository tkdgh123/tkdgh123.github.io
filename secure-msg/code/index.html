/* 덜미 클라이언트 (평문 버전): ./client 이름 [서버 주소]
   그냥 치면 전체 대화, /dm 이름 은 1:1, /all 로 돌아온다. /help 로 명령 보기.
   ※ 암호화는 뺀 단계다. 메시지는 평문으로 오간다 (암호화 버전은 saved/ 와
      git 브랜치 with-crypto 에 보관). 서버·net 계층은 그대로 쓴다. */
#define _GNU_SOURCE   /* wcwidth */
#include <stdio.h>
#include <stdlib.h>
#include <stdarg.h>
#include <string.h>
#include <locale.h>
#include <wchar.h>
#include <poll.h>
#include <unistd.h>
#include <termios.h>
#include <sys/ioctl.h>
#include "net.h"

#define PORT       9000
#define MAX_PEERS  16
#define INPUT_MAX  1000

enum { TO_ALL = 0, TO_ONE = 1 };   /* MSG 본문 첫 바이트: 전체 대화인지 1:1인지 */

struct peer { char name[NAME_LEN]; };

static char my_name[NAME_LEN];
static char dm_to[NAME_LEN];        /* 비어 있으면 전체 대화 */
static struct peer peers[MAX_PEERS];
static int npeers;

static struct conn srv;
static struct packet pkt, out;      /* 64KB가 넘어서 전역 */

/* ---------- 화면: 받은 메시지는 위로, 입력 줄은 늘 맨 아래 ---------- */

static int tty;                     /* 키보드로 실행했나 (파이프로 넣으면 0) */
static struct termios saved_tio;
static char input[INPUT_MAX + 1];
static size_t input_len;
static int drawn_rows;              /* 입력 줄이 화면에서 몇 줄 넘어갔나 */

static void prompt(char *buf, size_t size)
{
    if (dm_to[0])
        snprintf(buf, size, "%s→%s> ", my_name, dm_to);
    else
        snprintf(buf, size, "%s> ", my_name);
}

/* s를 cols칸 터미널에 찍으면 몇 줄 아래로 넘어가나.
   한글은 2칸이고, 줄 끝에 1칸만 남으면 터미널이 그 글자를 다음 줄로 넘긴다 */
static int rows_after(const char *s, int cols)
{
    mbstate_t st;
    wchar_t wc;
    size_t len = strlen(s), n;
    int x = 0, rows = 0;

    memset(&st, 0, sizeof(st));
    while (len > 0) {
        n = mbrtowc(&wc, s, len, &st);
        if (n == (size_t)-2)
            break;
        int cw = 1;
        if (n == (size_t)-1) {
            memset(&st, 0, sizeof(st));
            n = 1;
        } else if ((cw = wcwidth(wc)) < 0) {
            cw = 0;
        }
        if (x + cw > cols) {
            rows++;
            x = 0;
        }
        x += cw;
        s += n;
        len -= n;
    }
    return rows;
}

/* 화면 색 (터미널일 때만). 입력 프롬프트에는 쓰지 않는다 — rows_after가
   ESC 바이트를 칸으로 세어 줄 계산이 틀어진다. 메시지 줄에만 쓴다 */
#define C_ACCENT (tty ? "\033[36m" : "")
#define C_RESET  (tty ? "\033[0m"  : "")

static void clear_input(void)
{
    if (!tty)
        return;
    if (drawn_rows > 0)
        printf("\033[%dA", drawn_rows);   /* 입력 줄 첫 줄로 올라가서 */
    printf("\r\033[J");                     /* 그 아래를 다 지운다 */
}

static void draw_input(void)
{
    if (!tty)
        return;
    char line[64 + INPUT_MAX];
    struct winsize ws;
    int cols = 80;

    prompt(line, sizeof(line));
    strcat(line, input);
    printf("%s", line);
    if (ioctl(1, TIOCGWINSZ, &ws) == 0 && ws.ws_col > 0)
        cols = ws.ws_col;
    drawn_rows = rows_after(line, cols);
    fflush(stdout);
}

/* 입력 줄을 지우고 한 줄 찍은 다음 입력 줄을 다시 그린다. 경고는 빨갛게 */
static void put_line(int warning, const char *text)
{
    clear_input();
    if (warning && tty)
        printf("\033[31m%s\033[0m\n", text);
    else if (tty && text[0] == '[') {   /* 줄 맨 앞 [태그]를 강조 */
        const char *close = strchr(text, ']');
        if (close)
            printf("\033[36m%.*s\033[0m%s\n", (int)(close - text + 1), text, close + 1);
        else
            printf("%s\n", text);
    } else
        printf("%s\n", text);
    draw_input();
    fflush(stdout);
}

static void show(const char *fmt, ...)
{
    char text[2100];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(text, sizeof(text), fmt, ap);
    va_end(ap);
    put_line(0, text);
}

/* 남이 보낸 글에 섞인 제어 문자를 ?로. 터미널 조작 문자(ESC 등)를 막는다 */
static void clean(char *s)
{
    for (; *s; s++)
        if ((unsigned char)*s < 0x20 || *s == 0x7f)
            *s = '?';
}

/* ---------- 접속한 사람 목록 ---------- */

static int find_peer(const char *name)
{
    for (int i = 0; i < npeers; i++)
        if (strncmp(peers[i].name, name, NAME_LEN) == 0)
            return i;
    return -1;
}

static void add_peer(const char *name)
{
    if (name[0] == '\0' || strncmp(name, my_name, NAME_LEN) == 0
        || find_peer(name) >= 0 || npeers >= MAX_PEERS)
        return;   /* 나 자신은 목록에 넣지 않는다 (서버가 내 HELLO도 돌려준다) */
    strncpy(peers[npeers].name, name, NAME_LEN);
    npeers++;
}

static void remove_peer(const char *name)
{
    int i = find_peer(name);
    if (i < 0)
        return;
    peers[i] = peers[--npeers];
    if (strncmp(dm_to, name, NAME_LEN) == 0)
        dm_to[0] = '\0';   /* 상대가 나가면 전체 대화로 돌아온다 */
}

/* 서버가 보낸 SYS가 "이름 입장" / "이름 퇴장" 이면 목록을 고친다 */
static void sys_update(const char *text)
{
    size_t L = strlen(text);
    const char *in = " 입장", *out_ = " 퇴장";
    size_t li = strlen(in), lo = strlen(out_);
    char nm[NAME_LEN];
    if (L > li && strcmp(text + L - li, in) == 0 && L - li < NAME_LEN) {
        memcpy(nm, text, L - li);
        nm[L - li] = '\0';
        add_peer(nm);
    } else if (L > lo && strcmp(text + L - lo, out_) == 0 && L - lo < NAME_LEN) {
        memcpy(nm, text, L - lo);
        nm[L - lo] = '\0';
        remove_peer(nm);
    }
}

/* ---------- 받은 패킷 ---------- */

static void handle_packet(const struct packet *p)
{
    char from[NAME_LEN + 1];
    memcpy(from, p->from, NAME_LEN);
    from[NAME_LEN] = '\0';
    clean(from);

    if (p->type == T_HELLO) {           /* 접속할 때 이미 있던 사람들을 알게 됨 */
        add_peer(from);
    } else if (p->type == T_MSG) {
        static char text[MAX_BODY];
        if (p->body_len < 1)
            return;
        size_t n = p->body_len - 1;
        if (n > MAX_BODY - 1)
            n = MAX_BODY - 1;
        memcpy(text, p->body + 1, n);
        text[n] = '\0';
        clean(text);
        if (p->body[0] == TO_ONE)
            show("[%s → 나] %s", from, text);
        else
            show("[%s] %s", from, text);
    } else if (p->type == T_SYS) {
        static char text[MAX_BODY];
        size_t n = p->body_len < MAX_BODY ? p->body_len : MAX_BODY - 1;
        memcpy(text, p->body, n);
        text[n] = '\0';
        clean(text);
        show("[알림] %s", text);
        sys_update(text);
    }
    /* 평문 버전에는 금고·증명 같은 다른 type이 오지 않는다 */
}

/* ---------- 보내기 ---------- */

static void send_msg(uint8_t flag, const char *to, const char *text)
{
    memset(&out, 0, sizeof(out));
    out.type = T_MSG;
    strncpy(out.from, my_name, NAME_LEN);
    strncpy(out.to, to, NAME_LEN);
    size_t len = strlen(text);
    if (len > INPUT_MAX)
        len = INPUT_MAX;
    out.body[0] = flag;
    memcpy(out.body + 1, text, len);
    out.body_len = (uint32_t)(len + 1);
    send_packet(srv.fd, &out);
}

static void send_text(const char *text)
{
    if (dm_to[0]) {
        send_msg(TO_ONE, dm_to, text);
    } else if (npeers == 0) {
        show("[알림] 아직 아무도 없습니다");
    } else {
        for (int i = 0; i < npeers; i++)
            send_msg(TO_ALL, peers[i].name, text);
    }
}

static void cmd_who(void)
{
    char line[64 + MAX_PEERS * NAME_LEN];
    int off = snprintf(line, sizeof(line), "[접속 중] %s(나)", my_name);
    for (int i = 0; i < npeers; i++)
        off += snprintf(line + off, sizeof(line) - off, ", %s", peers[i].name);
    show("%s", line);
}

/* ---------- 입력 ---------- */

/* 친 줄을 화면에 남기고 입력 줄은 비운다 */
static void echo_line(const char *line)
{
    if (!tty)
        return;
    char p[64];
    prompt(p, sizeof(p));
    clear_input();
    printf("%s%s\n", p, line);
    draw_input();
    fflush(stdout);
}

static void on_line(const char *line)
{
    if (strcmp(line, "/quit") == 0) {
        clear_input();
        exit(0);
    } else if (strcmp(line, "/all") == 0) {
        dm_to[0] = '\0';
        show("[전체 대화] 접속한 사람 모두에게 갑니다");
    } else if (strncmp(line, "/dm ", 4) == 0) {
        const char *name = line + 4;
        if (strcmp(name, my_name) == 0 || !name[0]) {
            show("[1:1 대화] 쓰는 법: /dm 이름");
        } else if (find_peer(name) < 0) {
            show("[1:1 대화] %s는 접속해 있지 않습니다 (/who 로 확인)", name);
        } else {
            strncpy(dm_to, name, NAME_LEN);
            show("[1:1 대화] %s에게만 갑니다 · /all 로 돌아오기", name);
        }
    } else if (strcmp(line, "/who") == 0) {
        cmd_who();
    } else if (strcmp(line, "/help") == 0) {
        show("[알림] 그냥 치면 전체 대화 · /dm 이름 · /all · /who · /quit");
    } else if (line[0] == '/') {
        show("[알림] 모르는 명령입니다 · /help");
    } else {
        send_text(line);
    }
}

static void on_key(unsigned char ch)
{
    static int esc;
    if (esc == 1) {
        esc = (ch == '[') ? 2 : 0;
        return;
    }
    if (esc == 2) {
        if (ch >= 0x40 && ch <= 0x7e)
            esc = 0;
        return;
    }

    if (ch == 0x1b) {
        esc = 1;
    } else if (ch == '\n' || ch == '\r') {
        char line[INPUT_MAX + 1];
        if (input_len == 0)
            return;
        strcpy(line, input);
        input_len = 0;
        input[0] = '\0';
        echo_line(line);
        on_line(line);
    } else if (ch == 0x7f || ch == 0x08) {   /* Backspace: 한글은 이어지는 바이트를 한 번에 */
        while (input_len > 0 && (input[input_len - 1] & 0xC0) == 0x80)
            input_len--;
        if (input_len > 0)
            input_len--;
        input[input_len] = '\0';
    } else if (ch == 0x03 || (ch == 0x04 && input_len == 0)) {   /* Ctrl+C, 빈 줄 Ctrl+D */
        clear_input();
        exit(0);
    } else if (ch == 0x15) {   /* Ctrl+U: 줄 지우기 */
        input_len = 0;
        input[0] = '\0';
    } else if (ch >= 0x20) {
        int lead = (ch & 0xC0) != 0x80;
        if (input_len >= INPUT_MAX || (lead && input_len >= INPUT_MAX - 4))
            return;
        input[input_len++] = (char)ch;
        input[input_len] = '\0';
    }
}

static void restore_tty(void)
{
    if (tty)
        tcsetattr(0, TCSANOW, &saved_tio);
}

static void raw_tty(void)
{
    tty = isatty(0) && isatty(1);
    if (!tty)
        return;
    tcgetattr(0, &saved_tio);
    struct termios t = saved_tio;
    t.c_lflag &= ~(ICANON | ECHO | ISIG);
    t.c_cc[VMIN] = 1;
    t.c_cc[VTIME] = 0;
    tcsetattr(0, TCSANOW, &t);
    atexit(restore_tty);
}

static int valid_name(const char *s)
{
    size_t n = strlen(s);
    if (n == 0 || n >= NAME_LEN)
        return 0;
    for (; *s; s++)
        if (*s == ' ' || *s == '/' || (unsigned char)*s < 0x20)
            return 0;
    return 1;
}

int main(int argc, char **argv)
{
    setlocale(LC_ALL, "");
    if (argc < 2 || !valid_name(argv[1])) {
        fprintf(stderr, "사용법: %s 이름 [서버 주소]\n", argv[0]);
        fprintf(stderr, "이름은 15바이트까지(한글 5글자), 띄어쓰기와 / 없이\n");
        return 1;
    }
    strncpy(my_name, argv[1], NAME_LEN);
    const char *host = argc >= 3 ? argv[2] : "127.0.0.1";

    srv.fd = net_connect(host, PORT);
    if (srv.fd < 0)
        return 1;

    /* HELLO: 이름만 알린다 (본문은 쓰지 않는다 — 서버는 헤더의 이름만 본다) */
    memset(&out, 0, sizeof(out));
    out.type = T_HELLO;
    strncpy(out.from, my_name, NAME_LEN);
    out.body_len = HELLO_LEN;
    send_packet(srv.fd, &out);

    raw_tty();
    show("%s덜미%s · %s · %s 접속", C_ACCENT, C_RESET, my_name, host);
    show("대화는 그냥 입력 · /dm 이름 · /all · /who · /help · /quit");

    struct pollfd fds[2] = {
        { .fd = 0,      .events = POLLIN },
        { .fd = srv.fd, .events = POLLIN },
    };
    for (;;) {
        if (poll(fds, 2, -1) < 0)
            continue;

        if (fds[1].revents & (POLLIN | POLLHUP | POLLERR)) {
            int r;
            if (conn_read(&srv) <= 0) {
                show("[알림] 서버와 연결이 끊어졌습니다");
                return 1;
            }
            while ((r = conn_take(&srv, &pkt)) == 1)
                handle_packet(&pkt);
            if (r < 0) {
                show("[알림] 서버가 이상한 패킷을 보냈습니다");
                return 1;
            }
        }

        if (fds[0].revents & (POLLIN | POLLHUP)) {
            unsigned char buf[256];
            ssize_t n = read(0, buf, sizeof(buf));
            if (n <= 0) {
                clear_input();
                return 0;
            }
            for (ssize_t i = 0; i < n; i++)
                on_key(buf[i]);
            clear_input();
            draw_input();
        }
    }
}

/* 덜미 서버: 내용을 모른 채 to에게 전해 주는 중계소 */
#include <stdio.h>
#include <stdarg.h>
#include <string.h>
#include <signal.h>
#include <poll.h>
#include <unistd.h>
#include <sys/socket.h>
#include "net.h"

#define MAX_CLIENTS 16

enum { ATK_NONE, ATK_TAMPER, ATK_DROP };   /* 이 사람에게 가는 다음 MSG에 걸어 둔 공격 */

struct client {
    struct conn c;
    char    name[NAME_LEN];
    uint8_t hello[HELLO_LEN];
    int     registered;
    int     attack;
    struct packet last;      /* 이 사람에게 마지막으로 중계한 패킷 (replay용) */
    int     has_last;
};

/* 한 칸 더: 17번째의 HELLO까지 읽은 뒤에 "꽉 참"을 보내야 알림이 사라지지 않는다
   (안 읽은 데이터가 남은 채 close하면 RST가 나가서 받는 쪽이 알림을 못 읽는다) */
static struct client clients[MAX_CLIENTS + 1];
static int nclients;
static struct packet pkt;   /* 받은 패킷. 64KB가 넘어서 지역 변수로 두지 않는다 */
static struct packet out;   /* 서버가 만들어 보내는 패킷 (SYS, HELLO 전달) */

static const char *type_name(int type)
{
    static const char *names[] = { "?", "HELLO", "MSG", "VAULT", "OPEN_REQ", "SHARE", "PROOF", "SYS" };
    return (type >= T_HELLO && type <= T_SYS) ? names[type] : names[0];
}

static int find_client(const char *name)
{
    for (int i = 0; i < nclients; i++)
        if (clients[i].registered && strncmp(clients[i].name, name, NAME_LEN) == 0)
            return i;
    return -1;
}

/* i번에게 서버 알림 한 줄 */
static void send_sys(int i, const char *fmt, ...)
{
    va_list ap;
    memset(&out, 0, sizeof(out));
    out.type = T_SYS;
    strncpy(out.to, clients[i].name, NAME_LEN - 1);
    va_start(ap, fmt);
    int n = vsnprintf((char *)out.body, MAX_BODY, fmt, ap);
    va_end(ap);
    out.body_len = n < MAX_BODY ? n : MAX_BODY - 1;
    send_packet(clients[i].c.fd, &out);
}

/* owner의 HELLO를 i번에게 */
static void send_hello(int i, int owner)
{
    memset(&out, 0, sizeof(out));
    out.type = T_HELLO;
    memcpy(out.from, clients[owner].name, NAME_LEN);
    memcpy(out.to, clients[i].name, NAME_LEN);
    out.body_len = HELLO_LEN;
    memcpy(out.body, clients[owner].hello, HELLO_LEN);
    send_packet(clients[i].c.fd, &out);
}

static void add_client(int fd)
{
    if (fd < 0)
        return;
    if (nclients == MAX_CLIENTS + 1) {   /* 남는 칸까지 찼으면 말없이 닫는다 */
        close(fd);
        return;
    }
    memset(&clients[nclients], 0, sizeof(clients[0]));
    clients[nclients].c.fd = fd;
    nclients++;
}

/* i번을 지우고 마지막 칸을 i번 자리로 옮긴다 */
static void drop_client(int i)
{
    char name[NAME_LEN];
    int was_registered = clients[i].registered;
    memcpy(name, clients[i].name, NAME_LEN);

    close(clients[i].c.fd);
    clients[i] = clients[--nclients];

    if (was_registered) {
        printf("[leave] %s\n", name);
        for (int k = 0; k < nclients; k++)
            if (clients[k].registered)
                send_sys(k, "%s 퇴장", name);
    }
}

static int handle_hello(int i, const struct packet *p)
{
    if (clients[i].registered) {
        send_sys(i, "이미 HELLO를 보냈습니다");
        return 0;
    }
    if (p->body_len != HELLO_LEN) {
        send_sys(i, "HELLO 길이가 잘못됨");
        drop_client(i);
        return -1;
    }
    if (p->from[0] == '\0' || find_client(p->from) >= 0) {
        send_sys(i, "이름 사용 중: %s", p->from);
        drop_client(i);
        return -1;
    }
    int registered = 0;
    for (int k = 0; k < nclients; k++)
        registered += clients[k].registered;
    if (registered == MAX_CLIENTS) {
        send_sys(i, "서버가 꽉 찼습니다 (%d명)", MAX_CLIENTS);
        drop_client(i);
        return -1;
    }

    struct client *me = &clients[i];
    memcpy(me->name, p->from, NAME_LEN);
    memcpy(me->hello, p->body, HELLO_LEN);
    me->registered = 1;
    printf("[join ] %s (fd %d)\n", me->name, me->c.fd);

    for (int k = 0; k < nclients; k++) {
        if (k == i || !clients[k].registered)
            continue;
        send_hello(i, k);   /* 새 사람에게 기존 전원의 HELLO */
        send_hello(k, i);   /* 기존 전원에게 새 HELLO */
    }
    for (int k = 0; k < nclients; k++)
        if (clients[k].registered)
            send_sys(k, "%s 입장", me->name);
    return 0;
}

/* 서버의 눈: 중계할 때마다 한 줄. 서버에는 암호문만 보인다 */
static void show_eye(const struct packet *p)
{
    printf("[relay] %s → %s  %-8s seq=%-4llu %4uB ", p->from, p->to, type_name(p->type),
           (unsigned long long)p->seq, p->body_len);
    for (uint32_t k = 0; k < p->body_len && k < 8; k++)
        printf(" %02x", p->body[k]);
    printf(p->body_len > 8 ? " …\n" : "\n");
}

static int color;   /* 서버 창이 터미널이면 공격 줄을 빨갛게 */

static void show_attack(const char *fmt, ...)
{
    va_list ap;
    if (color)
        printf("\033[31m");

/* to에게 중계. 걸어 둔 공격이 있으면 중계 직전에 적용한다 */
static void relay(int to, struct packet *p)
{
    struct client *t = &clients[to];
    unsigned long long seq = p->seq;

    if (p->type == T_MSG && t->attack == ATK_DROP) {
        t->attack = ATK_NONE;
        show_attack("[drop]   %s → %s  MSG      seq=%llu  몰래 버림", p->from, p->to, seq);
        return;
    }
    if (p->type == T_MSG && t->attack == ATK_TAMPER && p->body_len > 0) {
        t->attack = ATK_NONE;
        p->body[p->body_len - 1] ^= 0x01;
        show_attack("[tamper] %s → %s  MSG      seq=%llu  마지막 바이트 ^= 0x01", p->from, p->to, seq);
    } else {
        show_eye(p);
    }
    send_packet(t->c.fd, p);   /* 실패해도 다음 poll에서 끊김으로 잡힌다 */
    t->last = *p;
    t->has_last = 1;
}

/* 패킷 하나 처리. 이 사람을 끊었으면 -1 */
static int handle_packet(int i, struct packet *p)
{
    if (!clients[i].registered && p->type != T_HELLO) {
        send_sys(i, "먼저 HELLO를 보내야 합니다");
        drop_client(i);
        return -1;
    }
    if (p->type == T_HELLO)
        return handle_hello(i, p);
    if (p->type < T_MSG || p->type > T_PROOF) {
        send_sys(i, "모르는 type: %d", p->type);
        return 0;
    }
    if (strncmp(p->from, clients[i].name, NAME_LEN) != 0) {
        send_sys(i, "보낸 사람 이름이 다릅니다: %s", p->from);   /* 남의 이름으로 보내기 차단 */
        return 0;
    }

    int to = find_client(p->to);
    if (to < 0) {
        send_sys(i, "%s는 접속해 있지 않습니다", p->to);
        return 0;
    }
    relay(to, p);
    return 0;
}

/* ---------- 공격 실험실: 서버 창에서 치는 명령 ---------- */

static void run_command(const char *line)
{
    char cmd[16], name[NAME_LEN];
    if (sscanf(line, "%15s %15s", cmd, name) != 2
        || (strcmp(cmd, "tamper") != 0 && strcmp(cmd, "replay") != 0 && strcmp(cmd, "drop") != 0)) {
        if (line[0] != '\0')
            printf("[공격] 명령: tamper 이름 · replay 이름 · drop 이름\n");
        return;
    }
    int t = find_client(name);
    if (t < 0) {
        printf("[공격] %s는 접속해 있지 않습니다\n", name);
        return;
    }

    struct client *c = &clients[t];
    if (strcmp(cmd, "tamper") == 0) {
        c->attack = ATK_TAMPER;
        show_attack("[공격] %s에게 가는 다음 MSG의 마지막 바이트를 바꿉니다", name);
    } else if (strcmp(cmd, "drop") == 0) {
        c->attack = ATK_DROP;
        show_attack("[공격] %s에게 가는 다음 MSG를 몰래 버립니다", name);
    } else if (!c->has_last) {
        printf("[공격] %s에게 아직 중계한 패킷이 없습니다\n", name);
    } else {
        send_packet(c->c.fd, &c->last);
        show_attack("[replay] %s → %s  %-8s seq=%llu  한 번 더 보냄", c->last.from, c->last.to,
                    type_name(c->last.type), (unsigned long long)c->last.seq);
    }
}

/* 키보드에서 읽은 만큼 모아서 줄마다 명령 실행. 키보드가 없으면 *fd = -1 */
static void read_command(int *fd)
{
    static char buf[256];
    static size_t len;
    ssize_t n = read(0, buf + len, sizeof(buf) - 1 - len);
    if (n <= 0) {   /* 백그라운드로 켜서 키보드가 없음: 더 지켜보지 않는다 */
        *fd = -1;
        return;
    }
    len += n;

    char *nl;
    while ((nl = memchr(buf, '\n', len)) != NULL) {
        *nl = '\0';
        run_command(buf);
        size_t used = nl - buf + 1;
        len -= used;
        memmove(buf, buf + used, len);
    }
    if (len == sizeof(buf) - 1)   /* 줄바꿈 없이 너무 길면 버린다 */
        len = 0;
}

int main(void)
{
    int lfd = net_listen(9000);
    if (lfd < 0)
        return 1;

    signal(SIGPIPE, SIG_IGN);
    setvbuf(stdout, NULL, _IOLBF, 0);   /* 파일로 돌려도 줄마다 바로 나오게 */
    color = isatty(1);
    printf("[서버] 9000번에서 기다리는 중 · 공격 명령: tamper 이름 · replay 이름 · drop 이름\n");

    int key_fd = 0;   /* 키보드 */
    struct pollfd fds[2 + MAX_CLIENTS + 1];
    for (;;) {
        int n = nclients;   /* 이번 바퀴에 지켜볼 사람 수 */
        fds[0] = (struct pollfd){ .fd = lfd,    .events = POLLIN };
        fds[1] = (struct pollfd){ .fd = key_fd, .events = POLLIN };   /* -1이면 poll이 건너뜀 */
        for (int i = 0; i < n; i++)
            fds[2 + i] = (struct pollfd){ .fd = clients[i].c.fd, .events = POLLIN };

        if (poll(fds, 2 + n, -1) < 0)
            continue;

        /* 뒤에서부터: 도중에 지워도 아직 안 본 칸이 밀려오지 않는다 */
        for (int i = n - 1; i >= 0; i--) {
            if (!(fds[2 + i].revents & (POLLIN | POLLHUP | POLLERR)))
                continue;
            if (conn_read(&clients[i].c) <= 0) {
                drop_client(i);
                continue;
            }
            int r;
            while ((r = conn_take(&clients[i].c, &pkt)) == 1)
                if (handle_packet(i, &pkt) < 0)
                    break;   /* 이미 drop_client(i) 됨 */
            if (r < 0)
                drop_client(i);   /* 길이가 잘못됨: 버퍼를 늘리지 않고 끊는다 */
        }
        if (fds[1].revents & (POLLIN | POLLHUP))
            read_command(&key_fd);
        if (fds[0].revents & POLLIN)
            add_client(accept(lfd, NULL, NULL));
    }
}

static int rows_after(const char *s, int cols)
{
    mbstate_t st;
    wchar_t wc;
    size_t len = strlen(s), n;
    int x = 0, rows = 0;

    memset(&st, 0, sizeof(st));
    while (len > 0) {
        n = mbrtowc(&wc, s, len, &st);
        if (n == (size_t)-2)
            break;
        int cw = 1;
        if (n == (size_t)-1) {
            memset(&st, 0, sizeof(st));
            n = 1;
        } else if ((cw = wcwidth(wc)) < 0) {
            cw = 0;
        }
        if (x + cw > cols) {
            rows++;
            x = 0;
        }
        x += cw;
        s += n;
        len -= n;
    }
    return rows;
}

static void host_handle(int i, const struct packet *p)
{

        }

        return;

}

        host_drop(i);
        return;

    while ((r = conn_take(&oconn[i], &pkt)) == 1)
        host_handle(i, &pkt);

size_t pack_packet(const struct packet *p, uint8_t *out)
{
    uint32_t len = HDR_LEN + p->body_len;
    uint32_t nlen = htonl(len);
    uint8_t *q = out;

    memcpy(q, &nlen, 4);             q += 4;
    *q++ = p->type;
    memcpy(q, p->from, NAME_LEN);    q += NAME_LEN;
    memcpy(q, p->to, NAME_LEN);      q += NAME_LEN;
    for (int i = 7; i >= 0; i--)      /* seq: 큰 자리부터 한 바이트씩 */
        *q++ = (uint8_t)(p->seq >> (i * 8));
    memcpy(q, p->nonce, NONCE_LEN);  q += NONCE_LEN;
    memcpy(q, p->body, p->body_len);
    return 4 + len;
}

int send_packet(int fd, const struct packet *p)
{
    static uint8_t out[PKT_MAX];
    size_t total = pack_packet(p, out), sent = 0;

    while (sent < total) {           /* send는 일부만 보낼 수 있다 */
        ssize_t n = send(fd, out + sent, total - sent, MSG_NOSIGNAL);
        if (n <= 0)
            return -1;
        sent += (size_t)n;
    }
    return 0;
}

int conn_take(struct conn *c, struct packet *p)

    memcpy(&len, c->buf, 4);

/* MK_n = BLAKE2b(키 = CK_n, 내용 = "msg") */
static void message_key(const uint8_t ck[KEY_LEN], uint8_t mk[KEY_LEN])
{
    crypto_generichash(mk, KEY_LEN, (const uint8_t *)"msg", 3, ck, KEY_LEN);
}

/* CK_n+1 = BLAKE2b(키 = CK_n, 내용 = "next" ‖ 평문_n). 옛 CK는 덮어써서 지운다 */
static void next_chain(uint8_t ck[KEY_LEN], const uint8_t *pt, size_t len)
{
    crypto_generichash_state st;
    uint8_t next[KEY_LEN];

    crypto_generichash_init(&st, ck, KEY_LEN, KEY_LEN);
    crypto_generichash_update(&st, (const uint8_t *)"next", 4);
    crypto_generichash_update(&st, pt, len);
    crypto_generichash_final(&st, next, KEY_LEN);
    memcpy(ck, next, KEY_LEN);
    sodium_memzero(next, KEY_LEN);
}

/* 평문을 잠그고 래칫을 한 칸 전진. 암호문 길이(평문 + 16)를 돌려준다 */
size_t chain_seal(struct chain *c, const uint8_t *ad, const uint8_t *pt, size_t len,
                  uint8_t nonce[24], uint8_t *ct)
{
    uint8_t mk[KEY_LEN];

    message_key(c->ck, mk);
    size_t n = aead_seal(mk, ad, AD_LEN, pt, len, nonce, ct);
    next_chain(c->ck, pt, len);
    sodium_memzero(mk, KEY_LEN);
    return n;
}

int chain_open(struct chain *c, const uint8_t *ad, const uint8_t nonce[24],
               const uint8_t *ct, size_t ct_len, uint8_t *pt)
{

    } else {
        int ok = read(fd, salt, SALT_LEN) == SALT_LEN && read(fd, nonce, 24) == 24
              && read(fd, check, CHECK_LEN) == CHECK_LEN;
        close(fd);
        for (int tries = 0; ok && tries < 3 && !on; tries++) {

/* AD = 대화 이름 ‖ 줄 번호 8바이트 */
static size_t make_ad(uint8_t *ad, const char *conv, uint64_t line_no)
{
    size_t n = strlen(conv);
    memcpy(ad, conv, n);
    for (int i = 7; i >= 0; i--)
        ad[n++] = (uint8_t)(line_no >> (i * 8));
    return n;
}

/* 줄 하나 읽기. 1 읽음, 0 파일 끝, -1 망가짐 */
static int read_record(int fd, uint8_t nonce[24], uint8_t *ct, uint32_t *ct_len)
{
    uint8_t len4[4];
    ssize_t n = read(fd, len4, 4);
    if (n == 0)
        return 0;
    *ct_len = (uint32_t)len4[0] << 24 | len4[1] << 16 | len4[2] << 8 | len4[3];
    if (n != 4 || *ct_len < TAG_LEN || *ct_len > MAX_LINE + TAG_LEN)
        return -1;
    if (read(fd, nonce, 24) != 24 || read(fd, ct, *ct_len) != (ssize_t)*ct_len)
        return -1;
    return 1;
}

void history_add(const char *conv, const char *line)
{
    static uint8_t rec[4 + 24 + MAX_LINE + TAG_LEN], ct[MAX_LINE + TAG_LEN];
    char path[600], name[64];
    uint8_t ad[64 + 8], nonce[24];
    uint32_t ct_len;
    uint64_t count = 0;
    size_t len = strlen(line);

    if (!on || len > MAX_LINE)
        return;
    snprintf(name, sizeof(name), "%s.log", conv);
    file_path(name, path, sizeof(path));
    int fd = open(path, O_RDWR | O_CREAT | O_APPEND, 0600);
    if (fd < 0)
        return;
    while (read_record(fd, nonce, ct, &ct_len) == 1)   /* 지금까지 몇 줄인지 = 새 줄 번호 */
        count++;

    size_t ad_len = make_ad(ad, conv, count);
    ct_len = (uint32_t)aead_seal(key, ad, ad_len, (const uint8_t *)line, len, rec + 4, rec + 4 + 24);
    rec[0] = (uint8_t)(ct_len >> 24);
    rec[1] = (uint8_t)(ct_len >> 16);
    rec[2] = (uint8_t)(ct_len >> 8);
    rec[3] = (uint8_t)ct_len;
    if (write(fd, rec, 4 + 24 + ct_len) < 0)
        perror("history");
    close(fd);
}
